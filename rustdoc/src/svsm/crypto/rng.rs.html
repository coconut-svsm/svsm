5//! SVSM Random Number Generator (RNG) management.
6//!
7//! Whenever users need an cryptographically secure RNG, they should invoke [`get_svsm_rng()`] to
8//! obtain an exclusively owned [`SvsmRng`] instance.
9

30/// Hash algorithm to be used for all NIST HashDrbg instantiations.
31// C.f. NIST SP 800-90A Rev. 1: the HashDrbg's security strength is equal to the underlying hash
32// algorithm's pre-image resistance. That is, SHA256 gives a security strength of 256 bits.
33const SVSM_RNG_DRBG_HASH_ALG: TpmiAlgHash = TpmiAlgHash::Sha256;
34
35/// Number of [`SVSM_RNG_POOL`] slots.
36const SVSM_RNG_POOL_SIZE: usize = 4;
37
38/// RNG instance pool
39///
40/// [`SvsmRng::drop()`] returns instances back into some free slot, if any. [`get_svsm_rng()`] tries
41/// to take a currently unused instance from the pool before attempting to instantiate one.
42#[allow(clippy::type_complexity)]
43static SVSM_RNG_POOL: SpinLock<
44    [Option<Box<rng::ChainedRng<rng::X86RdSeedRng, rng::HashDrbg>>>; SVSM_RNG_POOL_SIZE],
45> = SpinLock::new([None, None, None, None]);
46
47/// Counter used for the RNG instance's personalization.
48static SVSM_RNG_INSTANTIATION_ID: atomic::AtomicU64 = atomic::AtomicU64::new(0);
49
50/// Opaque type implementing [`RngCore`] as suitable for the SVSM environment and build
51/// configuration.
52///
53/// Instances of `SvsmRng` are to be obtained through [`get_svsm_rng()`].
54///
55/// No assumptions must be made about `SvsmRng`, other than instances thereof are small, typically
56/// of pointer size, and that it implements [`RngCore`].
57#[allow(missing_debug_implementations)]
58pub struct SvsmRng {
59    // Is never None while self is alive, the rng needs to get stored in an Option<> so that it can
60    // get returned back to the pool when dropped.
61    rng: Option<Box<rng::ChainedRng<X86RdSeedRng, rng::HashDrbg>>>,
62}
63
64impl SvsmRng {
65    fn instantiate() -> Result<Self, CryptoError> {
66        // The FixedVec internal representation is optimized for the case of power-of-two lengths,
67        // relative to a specified compile-time "base" value. In practice, the HashDrbg seed length
68        // is equal to the digest length for all known hashes, so use that for the FixedVec base. If
69        // not a power of two (e.g. SHA-1), the optimization wouldn't be possible anyway, so the
70        // value is irrelevant then.
71        const DRBG_HASH_ALG_DIGEST_LEN_LOG2: u32 =
72            hash_alg_digest_len(SVSM_RNG_DRBG_HASH_ALG).ilog2();
73        let mut hash_drbg_entropy = zeroize::Zeroizing::new(
74            FixedVec::<u8, DRBG_HASH_ALG_DIGEST_LEN_LOG2>::new_with_default(
75                HashDrbg::min_seed_entropy_len(SVSM_RNG_DRBG_HASH_ALG),
76            )
77            .map_err(|e| match e {
78                FixedVecMemoryAllocationFailure => CryptoError::MemoryAllocationFailure,
79            })?,
80        );
81
82        let mut rdseed_rng = X86RdSeedRng::instantiate().map_err(|_| CryptoError::RngFailure)?;
83        rdseed_rng.generate::<_, EmptyCryptoIoSlices>(
84            io_slices::SingletonIoSliceMut::new(hash_drbg_entropy.as_mut_slice())
85                .map_infallible_err(),
86            None,
87        )?;
88
89        let mut personalization = [0u8; 16];
90        personalization[..8].copy_from_slice(b"SVSM-RNG");
91        personalization[8..].copy_from_slice(
92            &SVSM_RNG_INSTANTIATION_ID
93                .fetch_add(1, atomic::Ordering::Relaxed)
94                .to_ne_bytes(),
95        );
96
97        let hash_drbg_rng = rng::HashDrbg::instantiate(
98            SVSM_RNG_DRBG_HASH_ALG,
99            &hash_drbg_entropy,
100            None,
101            Some(&personalization),
102        )?;
103
104        let chained_rng = rng::ChainedRng::chain(rdseed_rng, hash_drbg_rng);
105
106        Ok(Self {
107            rng: Some(box_try_new(chained_rng).map_err(|_| CryptoError::MemoryAllocationFailure)?),
108        })
109    }
110}
111
112impl rng::RngCore for SvsmRng {
113    fn generate<
114        'a,
115        'b,
116        OI: cocoon_tpm_crypto::CryptoWalkableIoSlicesMutIter<'a>,
117        AII: cocoon_tpm_crypto::CryptoPeekableIoSlicesIter<'b>,
118    >(
119        &mut self,
120        output: OI,
121        additional_input: Option<AII>,
122    ) -> Result<(), rng::RngGenerateError> {
123        let rng = match self.rng.as_mut() {
124            Some(rng) => rng,
125            None => {
126                return Err(rng::RngGenerateError::CryptoError(CryptoError::Internal));
127            }
128        };
129        rng.generate(output, additional_input)
130    }
131}
132
133impl Drop for SvsmRng {
134    fn drop(&mut self) {
135        if let Some(rng) = self.rng.take() {
136            // Return the rng instance in some free pool slot, if any.
137            let mut pool = SVSM_RNG_POOL.lock();
138            for pool_slot in pool.iter_mut() {
139                if pool_slot.is_none() {
140                    *pool_slot = Some(rng);
141                    break;
142                }
143            }
144        }
145    }
146}
147
148/// Obtain an exclusively owned [`SvsmRng`] instance.
149pub fn get_svsm_rng() -> Result<SvsmRng, CryptoError> {
150    let mut pool = SVSM_RNG_POOL.lock();
151    for pool_slot in pool.iter_mut() {
152        if let Some(rng) = pool_slot.take() {
153            return Ok(SvsmRng { rng: Some(rng) });
154        }
155    }
156
157    SvsmRng::instantiate()
158}
