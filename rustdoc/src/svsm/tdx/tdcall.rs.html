68#[bitfield(u64)]
69struct EptMappingInfo {
70    #[bits(12)]
71    pub flags: u64,
72    #[bits(52)]
73    pub page_frame_number: u64,
74}

76impl From<PageFrame> for EptMappingInfo {
77    fn from(frame: PageFrame) -> Self {
78        let (gpa, flags) = match frame {
79            PageFrame::Size4K(gpa) => (u64::from(gpa), 0),
80            PageFrame::Size2M(gpa) => (u64::from(gpa), 1),
81            PageFrame::Size1G(gpa) => (u64::from(gpa), 2),
82        };
83        Self::new()
84            .with_flags(flags)
85            .with_page_frame_number(gpa >> PAGE_SHIFT)
86    }
87}

89/// # Safety
90/// This function has the potential to zero arbitrary memory, so the caller is
91/// required to ensure that the supplied physical address range is appropriate
92/// for acceptance.
93unsafe fn tdg_mem_page_accept(frame: PageFrame) -> u64 {
94    loop {
95        // SAFETY: executing TDCALL requires the use of assembly.  The caller
96        // takes responsibility for correctness of the parameters.
97        let err = unsafe {
98            let mut ret: u64;
99            asm!("tdcall",
100                 in("rax") TDG_MEM_PAGE_ACCEPT,
101                 in("rcx") EptMappingInfo::from(frame).into_bits(),
102                 lateout("rax") ret,
103                 options(att_syntax));
104            ret
105        };
106        if !tdx_recoverable_error(err) {
107            return err;
108        }
109    }
110}

112/// Accepts the page of the given size at `addr`, zeroing it if it was
113/// already accepted.
114///
115/// # Safety
116///
117/// See `td_accept_physical_memory()`
118unsafe fn td_accept_physical_page(addr: PhysAddr, size: PageSize) -> Result<(), SvsmError> {
119    let frame = match size {
120        PageSize::Regular => PageFrame::Size4K(addr),
121        PageSize::Huge => PageFrame::Size2M(addr),
122    };
123    // SAFETY: the caller takes responsibility for the correct usage of the
124    // physical address.
125    let ret = unsafe { tdx_result(tdg_mem_page_accept(frame)) };
126    match ret {
127        Ok(TdxSuccess::PageAlreadyAccepted) => {
128            // The caller is expected not to accept a page twice unless doing
129            // so is known to be safe.  If the TDX module indicates that the
130            // page was already accepted, it must mean that the page was not
131            // removed after a previous attempt to accept.  In this case, the
132            // page must be zeroed now because the caller expects every
133            // accepted page to be zeroed.
134            let len = usize::from(size);
135            // SAFETY: the caller takes responsibility for the correct usage of
136            // the physical address.
137            unsafe {
138                let mapping = PerCPUPageMappingGuard::create(addr, addr + len, 0)?;
139                mapping.virt_addr().as_mut_ptr::<u8>().write_bytes(0, len);
140            }
141            Ok(())
142        }
143        Ok(_) => Ok(()),
144        Err(TdxError::PageSizeMismatch) if size == PageSize::Huge => {
145            // Process this 2 MB page as a series of 4 KB pages.
146            for addr in MemoryRegion::new(addr, PAGE_SIZE_2M).iter_pages(PageSize::Regular) {
147                // SAFETY: the caller takes responsibility for the correct
148                // usage of the physical address.
149                unsafe { td_accept_physical_page(addr, PageSize::Regular)? };
150            }
151            Ok(())
152        }
153        Err(e) => Err(e.into()),
154    }
155}

157/// # Safety
158/// This function will zero arbitrary memory, so the caller is required to
159/// ensure that the supplied physical address range is appropriate for
160/// acceptance.  The caller is additionally required to ensure that the address
161/// range is appropriate aligned to 4 KB boundaries.
162pub unsafe fn td_accept_physical_memory(region: MemoryRegion<PhysAddr>) -> Result<(), SvsmError> {
163    if !region.start().is_aligned(PAGE_SIZE) {
164        return Err(SvsmError::InvalidAddress);
165    }
166
167    for (addr, size) in region.iter_pages_largest() {
168        // SAFETY: the caller takes responsibility for the correct usage of the
169        // physical address.
170        unsafe { td_accept_physical_page(addr, size)? };
171    }
172
173    Ok(())
174}

176/// Accepts the page of the given size that `vaddr` maps to `paddr`, zeroing it
177/// if it was already accepted.
178///
179/// # Safety
180///
181/// See `td_accept_virtual_memory()`
182unsafe fn td_accept_virtual_page(
183    vaddr: VirtAddr,
184    paddr: PhysAddr,
185    size: PageSize,
186) -> Result<(), SvsmError> {
187    let frame = match size {
188        PageSize::Regular => PageFrame::Size4K(paddr),
189        PageSize::Huge => PageFrame::Size2M(paddr),
190    };
191    // SAFETY: the caller takes responsibility for the correct usage of the
192    // physical address.
193    let ret = unsafe { tdx_result(tdg_mem_page_accept(frame)) };
194    match ret {
195        Ok(TdxSuccess::PageAlreadyAccepted) => {
196            // Zero the page.
197            // SAFETY: the caller takes responsibility for the correct usage of
198            // the virtual address.
199            unsafe {
200                vaddr.as_mut_ptr::<u8>().write_bytes(0, usize::from(size));
201            }
202            Ok(())
203        }
204        Ok(_) => Ok(()),
205        Err(TdxError::PageSizeMismatch) if size == PageSize::Huge => {
206            // Process this 2 MB page as a series of 4 KB pages.
207            for offset in (0..PAGE_SIZE_2M).step_by(PAGE_SIZE) {
208                // SAFETY: the caller takes responsibility for the correct
209                // usage of the physical address.
210                unsafe {
211                    td_accept_virtual_page(vaddr + offset, paddr + offset, PageSize::Regular)?;
212                }
213            }
214            Ok(())
215        }
216        Err(e) => Err(e.into()),
217    }
218}

220/// # Safety
221/// This function will zero arbitrary memory, so the caller is required to
222/// ensure that the supplied virtual address range is appropriate for
223/// acceptance.  The caller is additionally required to ensure that the address
224/// range is appropriate aligned to 4 KB boundaries.
225pub unsafe fn td_accept_virtual_memory(region: MemoryRegion<VirtAddr>) -> Result<(), SvsmError> {
226    if !region.start().is_aligned(PAGE_SIZE) {
227        return Err(SvsmError::InvalidAddress);
228    }
229
230    for (vaddr, size) in region.iter_pages_largest() {
231        let frame = virt_to_frame(vaddr);
232        if frame.size() >= usize::from(size) {
233            // SAFETY: the caller takes responsibility for the correct usage of
234            // the virtual address.
235            unsafe { td_accept_virtual_page(vaddr, frame.address(), size)? };
236        } else {
237            // A 2 MB page mapped with 4 KB pages, which need not be physically
238            // contiguous.
239            for vaddr in MemoryRegion::new(vaddr, PAGE_SIZE_2M).iter_pages(PageSize::Regular) {
240                let paddr = virt_to_frame(vaddr).address();
241                // SAFETY: the caller takes responsibility for the correct
242                // usage of the virtual address.
243                unsafe { td_accept_virtual_page(vaddr, paddr, PageSize::Regular)? };
244            }
245        }
246    }
247    Ok(())
248}

250pub fn tdcall_get_ve_info() -> Option<TdVeInfo> {
251    let mut out_rcx: u64;
252    let mut out_rdx: u64;
253    let mut out_r8: u64;
254    let mut out_r9: u64;
255    let mut out_r10: u64;
256    // SAFETY: executing TDCALL requires the use of assembly.
257    let err = unsafe {
258        let mut ret: u64;
259        asm!("tdcall",
260             in("rax") TDG_VP_VEINFO_GET,
261             lateout("rax") ret,
262             out("rcx") out_rcx,
263             out("rdx") out_rdx,
264             out("r8") out_r8,
265             out("r9") out_r9,
266             out("r10") out_r10,
267             options(att_syntax));
268        ret
269    };
270    match tdx_result(err) {
271        Ok(_) => Some(TdVeInfo {
272            exit_reason: out_rcx as u32,
273            exit_qualification: out_rdx,
274            gla: out_r8,
275            gpa: out_r9,
276            exit_instruction_length: out_r10 as u32,
277            exit_instruction_info: ((out_r10 >> 32) as u32),
278        }),
279        Err(TdxError::NoVeInfo) => None,
280        Err(e) => panic!("Unknown TD error: {e:?}"),
281    }
282}

284pub fn tdcall_vm_read(field: u64) -> u64 {
285    let (val, err) = loop {
286        let mut val: u64;
287        // SAFETY: executing TDCALL requires the use of assembly.
288        let err = unsafe {
289            let mut ret: u64;
290            asm!("tdcall",
291                 in("rax") TDG_VM_RD,
292                 in("rcx") 0,
293                 in("rdx") field,
294                 lateout("rax") ret,
295                 lateout("rdx") _,
296                 out("r8") val,
297                 options(att_syntax));
298            ret
299        };
300        if !tdx_recoverable_error(err) {
301            break (val, err);
302        }
303    };
304    // Ignore errors here since the caller cannot handle them.
305    debug_assert!(tdx_result(err).is_ok());
306    // val = 0 in case of no success.
307    val
308}

310pub fn tdvmcall_map_gpa(mut gpa: u64, size: u64) -> Result<(), TdxError> {
311    let pass_regs = TdxRegs::R10 | TdxRegs::R11 | TdxRegs::R12 | TdxRegs::R13;
312    let end = gpa + size;
313    loop {
314        let mut ret: u64;
315        let mut vmcall_ret: u64;
316        let mut retry_gpa: u64;
317        // SAFETY: executing TDCALL requires the use of assembly.
318        unsafe {
319            asm!("tdcall",
320                 in("rax") TDG_VP_TDVMCALL,
321                 in("rcx") pass_regs.bits(),
322                 in("r10") 0,
323                 in("r11") TDVMCALL_MAP_GPA,
324                 in("r12") gpa,
325                 in("r13") end - gpa,
326                 lateout("rax") ret,
327                 lateout("r10") vmcall_ret,
328                 lateout("r11") retry_gpa,
329                 options(att_syntax));
330        }
331
332        debug_assert!(tdx_result(ret).is_ok());
333        let err = tdvmcall_result(vmcall_ret);
334
335        // If a retry was requested, then reissue the call as requested by the
336        // host.  No validation is performed on this value because it is
337        // being passed to an untrusted host.
338        if err != Err(TdxError::Vmcall(Retry)) {
339            return err;
340        }
341
342        gpa = retry_gpa;
343    }
344}

346pub fn tdvmcall_cpuid(cpuid_fn: u32, cpuid_subfn: u32) -> CpuidResult {
347    let pass_regs =
348        TdxRegs::R10 | TdxRegs::R11 | TdxRegs::R12 | TdxRegs::R13 | TdxRegs::R14 | TdxRegs::R15;
349    let mut ret: u64;
350    let mut vmcall_ret: u64;
351    let mut result_eax: u32;
352    let mut result_ebx: u32;
353    let mut result_ecx: u32;
354    let mut result_edx: u32;
355    // SAFETY: executing TDCALL requires the use of assembly.
356    unsafe {
357        asm!("tdcall",
358             in("rax") TDG_VP_TDVMCALL,
359             in("rcx") pass_regs.bits(),
360             in("r10") 0,
361             in("r11") TDVMCALL_CPUID,
362             in("r12") cpuid_fn,
363             in("r13") cpuid_subfn,
364             lateout("rax") ret,
365             lateout("r10") vmcall_ret,
366             lateout("r11") _,
367             lateout("r12") result_eax,
368             lateout("r13") result_ebx,
369             lateout("r14") result_ecx,
370             lateout("r15") result_edx,
371             options(att_syntax));
372    }
373    // r10 is expected to be TDG.VP.VMCALL_SUCCESS per the GHCI spec
374    // Make sure the result matches the expectation
375    debug_assert!(tdvmcall_result(vmcall_ret).is_ok());
376    debug_assert!(tdx_result(ret).is_ok());
377
378    CpuidResult {
379        eax: result_eax,
380        ebx: result_ebx,
381        ecx: result_ecx,
382        edx: result_edx,
383    }
384}

437#[derive(Clone, Copy, Debug)]
438pub enum TdpHaltInterruptState {
439    Enabled,
440    Disabled,
441}
442
443pub fn tdvmcall_halt(interrupt_state: TdpHaltInterruptState) {
444    let pass_regs = TdxRegs::R10 | TdxRegs::R11 | TdxRegs::R12;
445    let interrupts_blocked = match interrupt_state {
446        TdpHaltInterruptState::Enabled => 0,
447        TdpHaltInterruptState::Disabled => 1,
448    };
449    let mut ret: u64;
450    let mut vmcall_ret: u64;
451    // SAFETY: executing TDCALL requires the use of assembly.
452    unsafe {
453        asm!("pushfq",
454             "test %r12d, %r12d",
455             "jnz 1f",
456             "sti",
457             "1:",
458             "tdcall",
459             "popfq",
460             in("rax") TDG_VP_TDVMCALL,
461             in("rcx") pass_regs.bits(),
462             in("r10") 0,
463             in("r11") TDVMCALL_HLT,
464             in("r12") interrupts_blocked,
465             lateout("rax") ret,
466             lateout("r10") vmcall_ret,
467             lateout("r11") _,
468             lateout("r12") _,
469             options(att_syntax));
470    }
471    // r10 is expected to be TDG.VP.VMCALL_SUCCESS per the GHCI spec
472    // Make sure the result matches the expectation
473    debug_assert!(tdvmcall_result(vmcall_ret).is_ok());
474    debug_assert!(tdx_result(ret).is_ok());
475}

477/// R12 bitfield for `TDVMCALL_REPORT_FATAL_ERROR`.
478#[bitfield(u64)]
479struct FatalErrorR12 {
480    error_code: u32,
481    #[bits(31)]
482    ext_error_code: u32,
483    gpa_valid: bool,
484}
485
486pub fn tdvmcall_report_fatal_error() -> ! {
487    let pass_regs = TdxRegs::R10 | TdxRegs::R11 | TdxRegs::R12;
488    // Default error code (0)
489    let r12 = FatalErrorR12::new();
490
491    // This TDCALL is not supposed to return, but guard against
492    // a non-cooperating host with an infinite loop, ensuring
493    // execution never leaves this function.
494    loop {
495        // SAFETY: executing TDCALL requires the use of assembly.
496        unsafe {
497            asm!("tdcall",
498                 in("rax") TDG_VP_TDVMCALL,
499                 in("rcx") pass_regs.bits(),
500                 in("r10") 0,
501                 in("r11") TDVMCALL_REPORT_FATAL_ERROR,
502                 in("r12") r12.into_bits(),
503            );
504        }
505    }
506}

508fn tdvmcall_io(port: u16, data: u32, size: usize, write: bool) -> u32 {
509    let pass_regs =
510        TdxRegs::R10 | TdxRegs::R11 | TdxRegs::R12 | TdxRegs::R13 | TdxRegs::R14 | TdxRegs::R15;
511    let mut ret: u64;
512    let mut vmcall_ret: u64;
513    let mut output: u32;
514    // SAFETY: executing TDCALL requires the use of assembly.
515    unsafe {
516        asm!("tdcall",
517             in("rax") TDG_VP_TDVMCALL,
518             in("rcx") pass_regs.bits(),
519             in("r10") 0,
520             in("r11") TDVMCALL_IO,
521             in("r12") size,
522             in("r13") write as u32,
523             in("r14") port as u32,
524             in("r15") data,
525             lateout("rax") ret,
526             lateout("r10") vmcall_ret,
527             lateout("r11") output,
528             lateout("r12") _,
529             lateout("r13") _,
530             lateout("r14") _,
531             lateout("r15") _,
532             options(att_syntax));
533    }
534
535    // Ignore errors here.  The caller cannot handle them, and since the
536    // I/O operation was performed by an untrusted source, the error
537    // information is not meaningfully different than a maliciously unreliable
538    // operation.
539    debug_assert!(tdvmcall_result(vmcall_ret).is_ok());
540    debug_assert!(tdx_result(ret).is_ok());
541
542    output
543}

556pub fn tdvmcall_hyperv_hypercall(regs: &mut X86GeneralRegs) {
557    let pass_regs = TdxRegs::RDX | TdxRegs::R8 | TdxRegs::R10 | TdxRegs::R11;
558    let mut ret: u64;
559    let mut vmcall_ret: u64;
560    let mut hypercall_ret: u64;
561    // A Hyper-V hypercall uses VMCALL but does not use the standard GHCI
562    // convention.  The distinction is encoded in R10, which always passes
563    // zero for GHCI calls, and which passes the Hyper-V hypercall code (which
564    // is never zero) in the case of hypercalls.
565    debug_assert_ne!({ regs.rcx }, 0);
566    // SAFETY: executing TDCALL requires the use of assembly.
567    unsafe {
568        asm!("tdcall",
569             in("rax") TDG_VP_TDVMCALL,
570             in("rcx") pass_regs.bits(),
571             in("r10") regs.rcx,
572             in("rdx") regs.rdx,
573             in("r8") regs.r8,
574             lateout("rax") ret,
575             lateout("r10") vmcall_ret,
576             lateout("r11") hypercall_ret,
577             options(att_syntax));
578    }
579
580    // Ignore errors here.  The caller cannot handle them, and the final
581    // status is not trustworthy anyway.
582    debug_assert!(tdx_result(ret).is_ok());
583    debug_assert!(tdvmcall_result(vmcall_ret).is_ok());
584
585    regs.rax = hypercall_ret as usize;
