7//! Flat single-bit atomic bitmap allocator.
8//!
9//! Each bit is one slot (`0` = free, `1` = used). Allocation finds a word that
10//! is not all-ones, takes the complement to expose free bits, and claims the
11//! first free bit with a CAS. All operations are atomic and take `&self`.
12//!
13//! Stable Rust cannot size `[AtomicU64; f(N)]` from one const generic, so both
14//! `N` and `WORDS` are required. Prefer [`flat_bitmap_allocator!`](crate::flat_bitmap_allocator)
15//! to keep them in sync.
16
17use core::sync::atomic::{AtomicU64, Ordering};
18
19/// Build a [`FlatBitmapAllocator`] with capacity `$n`, deriving `WORDS` automatically.
20#[macro_export]
21macro_rules! flat_bitmap_allocator {
22    ($n:expr) => {
23        $crate::utils::flat_bitmap_allocator::FlatBitmapAllocator::<
24            {
25                const N: usize = $n;
26                N
27            },
28            {
29                const N: usize = $n;
30                N.div_ceil(64)
31            },
32        >
33    };
34}
35
36/// Flat atomic bitmap with exactly `N` bits.
37///
38/// Padding bits past `N` in the last word (when `N` is not a multiple of 64)
39/// are permanently marked used so they are never allocated.
40#[derive(Debug)]
41pub struct FlatBitmapAllocator<const N: usize, const WORDS: usize> {
42    words: [AtomicU64; WORDS],
43}
44
45impl<const N: usize, const WORDS: usize> FlatBitmapAllocator<N, WORDS> {
46    /// Number of allocatable bits.
47    pub const CAPACITY: usize = N;
48
49    /// Create an allocator with every bit free. Padding bits past `N` in the
50    /// last word are marked used so they are never allocated.
51    pub const fn new() -> Self {
52        let mut words = [const { AtomicU64::new(0) }; WORDS];
53        let rem = N % (u64::BITS as usize);
54        if rem != 0 {
55            // High bits past `N` must look "used" so alloc never returns them.
56            words[WORDS - 1] = AtomicU64::new(!((1u64 << rem) - 1));
57        }
58        Self { words }
59    }
60
61    /// Allocate one free bit. Returns its index, or `None` if exhausted.
62    pub fn alloc(&self) -> Option<usize> {
63        for (word_idx, word) in self.words.iter().enumerate() {
64            loop {
65                let cur = word.load(Ordering::Relaxed);
66                // Skip full words (no free bits).
67                if cur == u64::MAX {
68                    break;
69                }
70                // Complement: free bits become 1; first free = trailing_zeros.
71                let bit = (!cur).trailing_zeros() as usize;
72                let new = cur | (1u64 << bit);
73                if word
74                    .compare_exchange_weak(cur, new, Ordering::AcqRel, Ordering::Relaxed)
75                    .is_ok()
76                {
77                    let idx = word_idx * (u64::BITS as usize) + bit;
78                    debug_assert!(idx < N);
79                    return Some(idx);
80                }
81                // Lost the race; retry this word.
82            }
83        }
84        None
85    }
86
87    /// Free a previously allocated bit.
88    pub fn free(&self, bit: usize) {
89        assert!(bit < N);
90        let word_idx = bit / (u64::BITS as usize);
91        let mask = !(1u64 << (bit % (u64::BITS as usize)));
92        self.words[word_idx].fetch_and(mask, Ordering::Release);
93    }
94}

96#[cfg(test)]
97mod tests {
98    #[test]
99    fn alloc_and_free() {
100        let b = <crate::flat_bitmap_allocator!(128)>::new();
101        assert_eq!(b.alloc(), Some(0));
102        assert_eq!(b.alloc(), Some(1));
103        b.free(0);
104        assert_eq!(b.alloc(), Some(0));
105    }
106
107    #[test]
108    fn skips_full_words() {
109        let b = <crate::flat_bitmap_allocator!(128)>::new();
110        for _ in 0..64 {
111            assert!(b.alloc().is_some());
112        }
113        assert_eq!(b.alloc(), Some(64));
114    }
115
116    #[test]
117    fn exhausted() {
118        let b = <crate::flat_bitmap_allocator!(64)>::new();
119        for _ in 0..64 {
120            assert!(b.alloc().is_some());
121        }
122        assert_eq!(b.alloc(), None);
123    }
124
125    #[test]
126    fn partial_capacity_padding() {
127        let b = <crate::flat_bitmap_allocator!(100)>::new();
128        let mut n = 0;
129        while b.alloc().is_some() {
130            n += 1;
131        }
132        assert_eq!(n, 100);
133        assert_eq!(b.alloc(), None);
134    }
