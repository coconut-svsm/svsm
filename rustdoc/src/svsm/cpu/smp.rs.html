46#[derive(Debug)]
47pub struct ApStartContextRef {
48    mapping: PerCPUPageMappingGuard,
49}
50
51impl ApStartContextRef {
52    /// # Safety
53    /// The caller is required to specify a correct address for the AP start
54    /// context.
55    pub unsafe fn new(ap_start_context_addr: u32) -> Result<Self, SvsmError> {
56        let paddr = PhysAddr::from(u64::from(ap_start_context_addr));
57        let mapping = PerCPUPageMappingGuard::create_4k(paddr)?;
58        Ok(Self { mapping })
59    }
60
61    pub fn as_mut_ptr(&self) -> *mut ApStartContext {
62        self.mapping.virt_addr().as_mut_ptr::<ApStartContext>()
63    }
64}

66/// # Safety
67fn start_cpu(
68    platform: &dyn SvsmPlatform,
69    percpu_shared: &'static PerCpuShared,
70    ap_start_context_ref: Option<&ApStartContextRef>,
71) -> Result<(), SvsmError> {
72    let start_rip: u64 = start_ap_entry as *const () as usize as u64;
73    let percpu = PerCpu::alloc()?;
74    let pgtable = new_pgtable()?;
75    let (pgtable, context_switch_stack, init_shadow_stack, exception_stacks) =
76        percpu.setup(pgtable)?;
77
78    let stack_slot = context_switch_stack
79        .checked_sub(size_of::<*mut CpuLaunch>())
80        .unwrap();
81    let stack_slot_paddr = pgtable.phys_addr(stack_slot)?;
82    let stack_slot_vaddr = phys_to_virt(stack_slot_paddr);
83    let mut launch = Box::new(CpuLaunch::new_application(
84        percpu,
85        percpu_shared,
86        pgtable,
87        context_switch_stack,
88        init_shadow_stack,
89        exception_stacks,
90    ));
91    let initial_context = launch.get_initial_context(start_rip);
92    platform.prepare_cpu(&mut launch, &initial_context)?;
93    let svsm_vmsa = launch.svsm_vmsa_info();
94    let launch_ptr = Box::into_raw(launch);
95
96    // SAFETY: the stack slot is backed by the target CPU's private stack and
97    // is not visible to that CPU until the launch operation below. The AP
98    // entry shim consumes this pointer before using the stack.
99    unsafe {
100        stack_slot_vaddr
101            .as_mut_ptr::<*mut CpuLaunch>()
102            .write(launch_ptr);
103    }
104    fence(Ordering::Release);
105
106    if let Err(err) = platform.start_cpu(
107        percpu_shared,
108        &initial_context,
109        svsm_vmsa,
110        ap_start_context_ref,
111    ) {
112        // SAFETY: a failed platform launch did not release the target CPU, so
113        // ownership of the launch record remains with this CPU.
114        drop(unsafe { Box::from_raw(launch_ptr) });
115        return Err(err);
116    }
117
118    while !percpu_shared.is_online() {}
119    Ok(())
120}

122pub fn start_secondary_cpus(
123    platform: &dyn SvsmPlatform,
124    cpus: &[ACPICPUInfo],
125    ap_start_context_ref: Option<&ApStartContextRef>,
126) {
127    // Create the shared CPU structures for each application processor
128    // while still running single processor.  This ensures that the
129    // PERCPU_AREAS array is completely initialized before any additional
130    // processor starts running.
131    for c in cpus.iter().filter(|c| c.apic_id != 0 && c.enabled) {
132        // SAFETY: it is safe to create new shraed CPU areas because the system
133        // is still running single processor.
134        unsafe {
135            PERCPU_AREAS.create_new(c.apic_id);
136        }
137    }
138
139    // Now loop over all CPUs to start them.
140    for (cpu_index, percpu_shared) in PERCPU_AREAS.iter().enumerate() {
141        // CPU 0 is already running, so it doesn't require processing.
142        if cpu_index != 0 {
143            // If this is the first AP being started, then advise the TLB
144            // package that future TLB flushes will have to be done with SMP
145            // scope.
146            if cpu_index == 1 {
147                set_tlb_flush_smp();
148            }
149
150            log::info!(
151                "Launching AP {} with APIC-ID {}",
152                cpu_index,
153                percpu_shared.apic_id()
154            );
155            start_cpu(platform, percpu_shared, ap_start_context_ref).expect("failed");
156        }
157    }
158}

160unsafe extern "C" {
161    fn start_ap_indirect();
162    fn start_ap_entry();
163}
164
165global_asm!(
166    r#"
167        .section .text
168        .globl start_ap_indirect
169    start_ap_indirect:
170        /*
171         * %rdi stores the address of ApStartContext
172         * Load fields from the context structure
173         */
174        movq    (%rdi), %r8     /* CR0 */
175        movq    8(%rdi), %r9    /* CR3 */
176        movq    16(%rdi), %r10  /* CR4 */
177        movl    24(%rdi), %eax  /* Low bits of EFER */
178        movl    28(%rdi), %edx  /* High bits of EFER */
179        movq    32(%rdi), %r12  /* Start RIP */
180        movq    40(%rdi), %rsp  /* Initial RSP */
181
182        /* Switch to the target environment.  This will remove the transition
183         * environment and context structure from the address space. */
184        movq    %r8, %cr0
185        movq    %r10, %cr4
186        movl    $0xC0000080, %ecx   /* EFER */
187        wrmsr
188        movq    %r9, %cr3
189
190        /* Mark the next stack frame as the bottom frame */
191        xor     %rbp, %rbp
192
193        /* Begin execution from the starting RIP */
194        jmp     *%r12
195        int3
196
197        .globl start_ap_entry
198    start_ap_entry:
199        /* Consume the owned CpuLaunch pointer from the initial stack. */
200        popq    %rdi
201        call    start_ap
202        int3
203        "#,
204    options(att_syntax)
205);

207pub fn set_ap_start_context(
208    initial_context: &hyperv::HvInitialVpContext,
209    ap_start_context_ref: &ApStartContextRef,
210) {
211    // SAFETY: the safety of the AP start context ref was guaranteed when the
212    // object was created.
213    unsafe {
214        ap_start_context_ref.as_mut_ptr().write(ApStartContext {
215            cr0: initial_context.cr0.try_into().unwrap(),
216            cr3: initial_context.cr3.try_into().unwrap(),
217            cr4: initial_context.cr4.try_into().unwrap(),
218            efer: initial_context.efer.try_into().unwrap(),
219            start_rip: initial_context.rip.try_into().unwrap(),
220            rsp: initial_context.rsp.try_into().unwrap(),
221            initial_rip: start_ap_indirect as *const () as usize,
222        });
223    }
224}

226#[unsafe(no_mangle)]
227extern "C" fn start_ap(launch: *mut CpuLaunch) -> ! {
228    fence(Ordering::Acquire);
229    // SAFETY: the BSP created this pointer with Box::into_raw() and transfers
230    // ownership exactly once through the initial AP stack slot.
231    let launch = unsafe { Box::from_raw(launch) };
232    let percpu = launch.percpu();
233
234    initialize_cpu(*launch, &**SVSM_PLATFORM).expect("Failed to initialize AP");
235    load_static_idt();
236
237    if is_cet_ss_enabled() {
238        let ssp_token = initial_shadow_stack().unwrap();
239        enable_shadow_stacks!(ssp_token);
240    }
241
242    percpu
243        .setup_idle_task()
244        .expect("Failed to allocate idle task for AP");
245
246    // Send a life-sign
247    log::info!("CPU {} is online", this_cpu().get_cpu_index());
248
249    // Set CPU online so that BSP can proceed
250    this_cpu_shared().set_online();
251
252    sse_init();
253
254    // SAFETY: there is no current task running on this processor yet, so
255    // initializing the scheduler is safe.
256    unsafe {
257        schedule_init();
258    }
259
260    unreachable!("Road ends here!");
