74// PERCPU areas virtual addresses into shared memory
75pub static PERCPU_AREAS: PerCpuAreas = PerCpuAreas::new();
76
77// We use an UnsafeCell to allow for a static with interior mutability.
78// Normally, we would need to guarantee synchronization on the backing
79// datatype, but this is not needed because writes to the structure only occur
80// at initialization, from CPU 0, and reads of the Vec should only occur after
81// all writes are done.
82#[derive(Debug)]
83pub struct PerCpuAreas {
84    areas: UnsafeCell<Vec<&'static PerCpuShared>>,
85}
86
87// SAFETY: Any operation that can affect synchronization safety is declared as
88// unsafe, and therefore callers guarantee that synchronization is always
89// maintained. Also see comment above struct declaration.
90unsafe impl Sync for PerCpuAreas {}
91
92impl PerCpuAreas {
93    const fn new() -> Self {
94        Self {
95            areas: UnsafeCell::new(Vec::new()),
96        }
97    }
98
99    /// # Safety
100    /// The areas vector obtained here is not multi-thread safe, so the caller
101    /// must guarantee that is not used in a context that expects multi-thread
102    /// safety.
103    unsafe fn get_areas(&self) -> &Vec<&'static PerCpuShared> {
104        // SAFETY: the caller guarantees that accessing the unsafe cell is
105        // appropriate.
106        unsafe { &*self.areas.get() }
107    }
108
109    /// # Safety
110    /// This function can only be invoked before any other processors have
111    /// started.  Otherwise, accesses to the areas vector will not be safe.
112    pub unsafe fn create_new(&self, apic_id: u32) -> &'static PerCpuShared {
113        // SAFETY: the caller guarantees that it is safe to obtain a mutable
114        // reference to the areas vector.
115        let areas = unsafe { &mut *self.areas.get() };
116
117        // Allocate a new shared per-CPU area to describe the APIC ID being
118        // created.  The CPU index will be the next in sequence ased on the
119        // size of the vector.  The new shared per-CPU area is allocated on
120        // the heap so it is globally visible.
121        let cpu_index = areas.len();
122        let percpu_shared = Box::leak(Box::new(PerCpuShared::new(apic_id, cpu_index)));
123
124        // Leak the box so the allocation persists with a static lifetime,
125        // and install the allocated per-CPU area into the areas vector.
126        areas.push(percpu_shared);
127
128        percpu_shared
129    }
130
131    pub fn len(&self) -> usize {
132        // SAFETY: it is safe to obtain a shared reference to the areas
133        // vector because callers attempting to mutate the vector guarantee
134        // that they cannot race with iteration.
135        let areas = unsafe { self.get_areas() };
136        areas.len()
137    }
138
139    pub fn is_empty(&self) -> bool {
140        // SAFETY: it is safe to obtain a shared reference to the areas
141        // vector because callers attempting to mutate the vector guarantee
142        // that they cannot race with iteration.
143        let areas = unsafe { self.get_areas() };
144        areas.is_empty()
145    }
146
147    pub fn iter(&self) -> Iter<'_, &'static PerCpuShared> {
148        // SAFETY: it is safe to obtain a shared reference to the areas
149        // vector because callers attempting to mutate the vector guarantee
150        // that they cannot race with iteration.
151        let areas = unsafe { self.get_areas() };
152        areas.iter()
153    }
154
155    // Fails if no such area exists or its address is NULL
156    pub fn get_by_apic_id(&self, apic_id: u32) -> Option<&'static PerCpuShared> {
157        // For this to not produce UB the only invariant we must
158        // uphold is that there are no mutations or mutable aliases
159        // going on when casting via as_ref(). This only happens via
160        // Self::push(), which is intentionally unsafe and private.
161        self.iter()
162            .find(|shared| shared.apic_id() == apic_id)
163            .copied()
164    }
165
166    /// Callers are expected to specify a valid CPU index.
167    pub fn get_by_cpu_index(&self, index: usize) -> &'static PerCpuShared {
168        // SAFETY: it is safe to obtain a shared reference to the areas
169        // vector because callers attempting to mutate the vector guarantee
170        // that they cannot race with iteration.
171        let areas = unsafe { self.get_areas() };
172        areas[index]
173    }
174}

239        // address can never overflow.
240        unsafe { Some(NonNull::new_unchecked(ptr)) }
241    }

320    pub fn request_ipi(&self, vector: u8) {
321        let index = vector >> 5;
322        let bit = 1u32 << (vector & 31);
323        // Request the IPI via the IRR vector before signaling that an IPI has
324        // been requested.
325        self.ipi_irr[index as usize].fetch_or(bit, Ordering::Relaxed);
326        self.ipi_pending.store(true, Ordering::Release);
327    }

331        self.ipi_pending.store(true, Ordering::Release);
332    }

367/// CPU-local data.
368///
369/// While the contents of this struct are never accessed outside the local CPU
370/// (except for the [`PerCpuShared`] portion), reentrant access to the structure
371/// from preempted context essentially behaves as concurrent access, since
372/// the preemption may happen at any instruction boundary. If we use interior
373/// mutability types, the reentrant access may lead to undefined behavior.
374/// Thus, we require that the type is [`Sync`]. This is currently enforced by
375/// protecting mutable fields with a [`RWLock`]. To avoid deadlocks, only the
376/// non-blocking methods in the lock are used, causing an immediate panic when a
377/// reentrant access is attempted.
378///
379/// If the [`Sync`] requirement below breaks your build, it means you introduced
380/// interior mutability, which is not safe.
381#[derive(Debug)]
382pub struct PerCpu
383where
384    Self: Sync,
385{
386    /// Linker-defined CPU-local variables accessed through `%GS`.
387    pub(super) percpu_area: PerCpuArea,
388
389    /// PerCpu Virtual Memory Range
390    vm_range: VMR,
391}

432    pub fn init_page_table(&self) -> Result<(), SvsmError> {
433        // SAFETY: The per-CPU address range is fully aligned to top-level
434        // paging boundaries.
435        unsafe {
436            self.vm_range.initialize()?;
437        }
438        Ok(())

441    fn allocate_stack(&self, base: VirtAddr) -> Result<VirtAddr, SvsmError> {
442        let stack = VMKernelStack::new()?;
443        let top_of_stack = (base + stack.top_of_stack()).align_down(16);
444
445        self.vm_range.insert_at(base, Arc::new(stack))?;
446
447        Ok(top_of_stack)
448    }
449
450    fn allocate_shadow_stack(
451        &self,
452        base: VirtAddr,
453        init: ShadowStackInit,
454    ) -> Result<VirtAddr, SvsmError> {
455        let shadow_stack = VMKernelStack::new_shadow()?;
456        let offset = shadow_stack.top_of_stack();
457        let shadow_stack_page = shadow_stack.shadow_page();
458        let shadow_stack_base = self.vm_range.insert_at(base, Arc::new(shadow_stack))?;
459        let (_, ssp) = init_shadow_stack(&shadow_stack_page, shadow_stack_base + offset, init);
460        Ok(ssp)
461    }

515    pub fn setup(
516        &self,
517        mut pgtable: PageBox<PageTable>,
518    ) -> Result<
519        (
520            PageBox<PageTable>,
521            VirtAddr,
522            Option<VirtAddr>,
523            ExceptionStacks,
524        ),
525        SvsmError,
526    > {
527        self.init_page_table()?;
528
529        // Map PerCpu data in own page-table
530        self.map_self()?;
531
532        // Allocate the linker-defined CPU-local data in this CPU's VMR.
533        self.percpu_area.map(self)?;
534
535        // Reserve ranges and initialize allocator for temporary mappings
536        self.initialize_vm_ranges()?;
537
538        let init_shadow_stack = cpu_has_feat(Feature::CetSS)
539            .then(|| self.allocate_init_shadow_stack())
540            .transpose()?;
541
542        // Allocate per-cpu context switch stack
543        let context_switch_stack = self.allocate_context_switch_stack()?;
544
545        if cpu_has_feat(Feature::CetSS) {
546            self.allocate_context_switch_shadow_stack()?;
547        }
548
549        // Allocate IST stacks
550        let double_fault = self.allocate_ist_stacks()?;
551
552        let double_fault_shadow = cpu_has_feat(Feature::CetSS)
553            .then(|| self.allocate_isst_shadow_stacks())
554            .transpose()?;
555
556        self.finish_page_table(&mut pgtable);
557
558        Ok((
559            pgtable,
560            context_switch_stack,
561            init_shadow_stack,
562            ExceptionStacks {
563                double_fault,
564                double_fault_shadow,
565            },
566        ))
567    }

569    pub(in crate::cpu) fn setup_percpu_keys(
570        &self,
571        shared: &'static PerCpuShared,
572        pgtable: &'static mut PageTable,
573        context_switch_stack: VirtAddr,
574        init_shadow_stack: Option<VirtAddr>,
575        exception_stacks: ExceptionStacks,
576    ) {
577        let initialize_keys = self.percpu_area.initialize();
578        self.percpu_area.load();
579        if !initialize_keys {
580            return;
581        }
582
583        init_percpu_shared(shared);
584
585        // Publish values prepared by the boot CPU now that this CPU can
586        // access its linker-backed per-CPU keys through %gs.
587        init_percpu_pgtable(pgtable);
588        init_context_switch_stack(context_switch_stack);
589        init_current_stack();
590        init_initial_shadow_stack(init_shadow_stack);
591        init_double_fault_stack(Some(exception_stacks.double_fault));
592        init_double_fault_shadow_stack(exception_stacks.double_fault_shadow);
593        setup_tss(double_fault_stack().unwrap());
594        init_isst(double_fault_shadow_stack());
595        init_vrange_4k();
596        init_vrange_2m();
597    }

599    // Setup code which needs to run on the target CPU
600    pub(in crate::cpu) fn setup_on_cpu(
601        &self,
602        platform: &dyn SvsmPlatform,
603    ) -> Result<(), SvsmError> {
604        // System calls from user mode rely on the syscall instruction
605        // raising #UD, see ex_handler_invalid_opcode().
606        disable_syscall_insn();
607
608        // Initialize platform resources before allocating shared pages.
609        platform.setup_percpu_current()?;
610
611        // The BSP allocates these later, when Hyper-V is initialized.
612        if self.shared().cpu_index() != 0 && cpu_has_feat(Feature::HyperV) {
613            allocate_hypercall_pages()?;
614        }
615        assert!(apic_id() == self.get_apic_id());
616        Ok(())
617    }

646    /// Fill in the initial context structure for the SVSM.
647    pub(in crate::cpu) fn get_initial_context(
648        &self,
649        start_rip: u64,
650        pgtable: &PageTable,
651        context_switch_stack: VirtAddr,
652    ) -> hyperv::HvInitialVpContext {
653        let data_segment = svsm_data_segment();
654        let mut gs_segment = data_segment;
655        gs_segment.base = self.percpu_area.gs_base() as u64;
656
657        hyperv::HvInitialVpContext {
658            rip: start_rip,
659            // Use the context switch stack as its initial stack. This stack
660            // will later serve as the transitory stack during context
661            // switching. It is accessible after switching to the first task's
662            // page table because it resides in the PerCpu virtual range.
663            //
664            // See switch_to() for details.
665            rsp: context_switch_stack
666                .checked_sub(size_of::<*mut CpuLaunch>())
667                .unwrap()
668                .into(),
669            rflags: 2,
670
671            cs: svsm_code_segment(),
672            ss: data_segment,
673            ds: data_segment,
674            es: data_segment,
675            fs: data_segment,
676            gs: gs_segment,
677            tr: self.svsm_tr_segment(),
678
679            gdtr: svsm_gdt_segment(),
680            idtr: svsm_idt_segment(),
681
682            cr0: read_cr0().bits(),
683            cr3: pgtable.cr3_value().into(),
684            cr4: read_cr4().bits(),
685            efer: read_efer().bits(),
686            pat: 0x0007040600070406u64,
687
688            ..Default::default()
689        }
690    }

711    pub fn alloc_guest_vmsa(&self) -> Result<PhysAddr, SvsmError> {
712        let use_alternate_injection = init_apic_emulation()?;

739    pub fn update_guest_mappings(&self) -> Result<(), SvsmError> {
740        let mut locked = self.guest_vmsa_ref();
741        let mut ret = Ok(());
742
743        if !locked.needs_update() {
744            // If there is no VMSA, then the update request must be considered a
745            // failure even though no work was required.
746            return match locked.vmsa_phys() {
747                Some(_) => Ok(()),
748                None => Err(SvsmError::MissingVMSA),
749            };
750        }
751
752        self.unmap_guest_vmsa();
753        self.unmap_caa();
754
755        match locked.vmsa_phys() {
756            Some(paddr) => self.map_guest_vmsa(paddr)?,
757            None => ret = Err(SvsmError::MissingVMSA),
758        }
759
760        if let Some(paddr) = locked.caa_phys() {
761            self.map_guest_caa(paddr.page_align())?
762        }
763
764        locked.set_updated();
765
766        ret
767    }

827pub fn this_cpu() -> &'static PerCpu {
828    // SAFETY: The PerCPU area is always mapped at the same virtual address, so
829    // dereferencing a pointer to that address is safe. The PerCPU area is also
830    // never freed, so using a static lifetime is safe as well.
831    unsafe { &*SVSM_PERCPU_BASE.as_ptr::<PerCpu>() }
832}
833
834pub fn try_this_cpu() -> Option<&'static PerCpu> {
835    let mut rcx: u64;
836
837    // SAFETY: Inline assembly to test the validity of the PerCpu page.
838    // Any #PF generated here is handled by the #PF handler and it will
839    // either return an error in %rcx or crash the system. This however
840    // does not impact any state related to memory safety.
841    unsafe {
842        asm!("1: movq ({0}), {1}",
843             "   xorq %rcx, %rcx",
844             "2:",
845             ".pushsection \"__early_exception_table\",\"a\"",
846             ".balign 16",
847             ".quad (1b)",
848             ".quad (2b)",
849             ".popsection",
850                in(reg) SVSM_PERCPU_BASE.bits(),
851                out(reg) _,
852                out("rcx") rcx,
853                options(att_syntax, nostack));
854    }
855    if rcx == 0 { Some(this_cpu()) } else { None }
856}

882#[derive(Debug)]
883pub struct PerCpuVmsas {
884    vmsas: RWLock<Vec<VmsaRegistryEntry>>,
885}
886
887impl PerCpuVmsas {
888    const fn new() -> Self {
889        Self {
890            vmsas: RWLock::new(Vec::new()),
891        }
892    }
893
894    pub fn exists(&self, paddr: PhysAddr) -> bool {
895        self.vmsas
896            .lock_read()
897            .iter()
898            .any(|vmsa| vmsa.paddr == paddr || vmsa.caa == paddr)
899    }
900
901    pub fn vmsa_exists(&self, paddr: PhysAddr) -> bool {
902        self.vmsas
903            .lock_read()
904            .iter()
905            .any(|vmsa| vmsa.paddr == paddr && vmsa.in_use)
906    }
907
908    pub fn overlaps(&self, region: &MemoryRegion<PhysAddr>) -> bool {
909        self.vmsas.lock_read().iter().any(|vmsa| {
910            region.overlap(&MemoryRegion::new(vmsa.paddr, PAGE_SIZE))
911                || region.overlap(&MemoryRegion::new(vmsa.caa, 8))
912        })
913    }
914
915    pub fn register(
916        &self,
917        paddr: PhysAddr,
918        caa: PhysAddr,
919        cpu_index: usize,
920        guest_owned: bool,
921    ) -> Result<(), SvsmError> {
922        let mut guard = self.vmsas.lock_write();
923        let vmsa_region = MemoryRegion::new(paddr, PAGE_SIZE);
924        let caa_region = MemoryRegion::new(caa, 8);
925        if guard.iter().any(|vmsa| {
926            let vr = MemoryRegion::new(vmsa.paddr, PAGE_SIZE);
927            let cr = MemoryRegion::new(vmsa.caa, 8);
928            vr.overlap(&vmsa_region)
929                || vr.overlap(&caa_region)
930                || cr.overlap(&vmsa_region)
931                || cr.overlap(&caa_region)
932        }) {
933            return Err(SvsmError::InvalidAddress);
934        }
935
936        guard.push(VmsaRegistryEntry::new(paddr, caa, cpu_index, guest_owned));
937        Ok(())
938    }
939
940    pub fn set_used(&self, paddr: PhysAddr) -> Option<usize> {
941        self.vmsas
942            .lock_write()
943            .iter_mut()
944            .find(|vmsa| vmsa.paddr == paddr && !vmsa.in_use)
945            .map(|vmsa| {
946                vmsa.in_use = true;
947                vmsa.cpu_index
948            })
949    }
950
951    pub fn remap_ca(&self, old_caa: PhysAddr, new_caa: PhysAddr) -> Result<(), SvsmError> {
952        let mut guard = self.vmsas.lock_write();
953        if guard.iter().any(|vmsa| {
954            let r1 = MemoryRegion::new(vmsa.paddr, PAGE_SIZE);
955            vmsa.caa == new_caa || r1.overlap(&MemoryRegion::new(new_caa, 8))
956        }) {
957            return Err(SvsmError::InvalidAddress);
958        }
959        guard
960            .iter_mut()
961            .find(|vmsa| vmsa.caa == old_caa)
962            .map(|vmsa| {
963                vmsa.caa = new_caa;
964                new_caa
965            });
966        Ok(())
967    }
968
969    pub fn unregister(
970        &self,
971        paddr: PhysAddr,
972        in_use: bool,
973    ) -> Result<VmsaRegistryEntry, SvsmError> {
974        let mut guard = self.vmsas.lock_write();
975        let index = guard
976            .iter()
977            .position(|vmsa| vmsa.paddr == paddr && vmsa.in_use == in_use)
978            .ok_or(SvsmError::InvalidParameter)?;
979
980        if in_use {
981            let vmsa = &guard[index];
982
983            let target_cpu = PERCPU_AREAS.get_by_cpu_index(vmsa.cpu_index);
984            target_cpu.clear_guest_vmsa_if_match(paddr);
985        }
986
987        Ok(guard.swap_remove(index))
988    }
989}

995pub fn cpu_idle_loop(cpu_index: usize) {
996    debug_assert_eq!(cpu_index, this_cpu().get_cpu_index());
997
998    loop {
999        // Go idle
1000        scheduler_idle();
1001
1002        // If idle was explicitly requested by another task, then schedule that
1003        // task to execute again in case it wants to perform processing as a
1004        // result of the wake from idle.
1005        let maybe_task = this_cpu().runqueue_mut().wake_from_idle();
1006        if let Some(task) = maybe_task {
1007            wake_and_schedule_task(task);
1008        }
1009
1010        // Execute any tasks that are currently runnable.
1011        schedule();
1012    }
