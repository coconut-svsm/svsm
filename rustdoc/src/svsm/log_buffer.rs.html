20fn print_warning(_e: SvsmError) {
21    _print(format_args!("ERROR: logging failed\n"));
22}

24/// Size of the early log buffer. One page is plenty: only a handful of
25/// messages are emitted before the regular log buffer takes over.
26const EARLY_LOG_SIZE: usize = PAGE_SIZE;
27
28/// Prefix added to messages stored in the early log buffer. It matches the
29/// prefix the idle task uses once the regular log buffer has taken over, so
30/// that the log reads continuously across the hand-over.
31const EARLY_LOG_PREFIX: &str = "[SVSM] ";
32
33/// Appended to the early log buffer contents when messages were dropped
34/// because the buffer ran out of space.
35const EARLY_LOG_TRUNCATED: &str = "[SVSM] <early log truncated>\n";
36
37/// A fixed size buffer holding the log messages emitted before the file
38/// system is up and the regular log buffer can be used. It needs no heap and
39/// no per-CPU state, so it is usable from the very beginning of kernel
40/// startup. [`switch_to_log_buffer()`] copies its contents to the regular log
41/// buffer, after which it is never used again.
42#[derive(Debug)]
43struct EarlyLogBuffer {
44    buffer: [u8; EARLY_LOG_SIZE],
45    fill: usize,
46    /// Set when messages were dropped because the buffer is full.
47    truncated: bool,
48}
49
50impl EarlyLogBuffer {
51    const fn new() -> Self {
52        Self {
53            buffer: [0u8; EARLY_LOG_SIZE],
54            fill: 0,
55            truncated: false,
56        }
57    }
58
59    /// Returns the bytes stored so far.
60    fn contents(&self) -> &[u8] {
61        &self.buffer[..self.fill]
62    }
63}
64
65impl fmt::Write for EarlyLogBuffer {
66    /// Keeps the earliest logs for startup diagnostics, dropping excess bytes
67    /// and flagging truncation.
68    fn write_str(&mut self, s: &str) -> fmt::Result {
69        let src = s.as_bytes();
70        let len = core::cmp::min(src.len(), EARLY_LOG_SIZE - self.fill);
71        self.buffer[self.fill..self.fill + len].copy_from_slice(&src[..len]);
72        self.fill += len;
73
74        if len != src.len() {
75            // End the truncated message with a newline so subsequent output
76            // starts on a separate line.
77            self.truncated = true;
78            self.buffer[EARLY_LOG_SIZE - 1] = b'\n';
79        }
80
81        Ok(())
82    }
83}
84
85static EARLY_LOG: SpinLock<EarlyLogBuffer> = SpinLock::new(EarlyLogBuffer::new());

87struct LogBuffer {}
88
89impl LogBuffer {
90    const fn new() -> Self {
91        Self {}
92    }
93}
94
95impl fmt::Write for LogBuffer {
96    /// Writes all bytes from the slice to the log buffer
97    fn write_str(&mut self, s: &str) -> fmt::Result {
98        let obj_handle = ObjHandle::new(0);
99        let buf = SliceRefBuffer::new(s.as_bytes());
100        let fs = obj_get(obj_handle).map_err(print_warning).unwrap();
101        let fs = fs
102            .as_fs()
103            .ok_or(SvsmError::LogError)
104            .map_err(print_warning)
105            .unwrap();
106        let _ = fs.write_buffer(&buf).map_err(print_warning).unwrap();
107        Ok(())
108    }
109}

111static LOGGER: ImmutAfterInitCell<SpinLock<LogBuffer>> = ImmutAfterInitCell::uninit();
112
113/// Stores a message in the early log buffer. There is no task to take a name
114/// from this early, so every message gets the same fixed prefix.
115///
116/// The message is dropped if the buffer is already locked further down the
117/// call stack, for instance by an exception handler interrupting a write. No
118/// IRQ-safe lock is available before per-CPU state exists, so waiting for the
119/// lock here would deadlock the only CPU running at this point.
120fn early_log_write(args: fmt::Arguments<'_>) {
121    if let Some(mut early) = EARLY_LOG.try_lock() {
122        // Writing to the early log buffer cannot fail.
123        early
124            .write_fmt(format_args!("{EARLY_LOG_PREFIX}{args}"))
125            .unwrap();
126    }
127}
128
129fn lb_write(args: fmt::Arguments<'_>) {
130    match LOGGER.try_get_inner() {
131        Ok(logger) => logger.lock().write_fmt(args).unwrap(),
132        Err(_) => early_log_write(args),
133    }
134}

136#[derive(Clone, Copy, Debug)]
137struct BufferLogger;
138
139impl log::Log for BufferLogger {
140    fn enabled(&self, _metadata: &log::Metadata<'_>) -> bool {
141        true
142    }
143
144    fn log(&self, record: &log::Record<'_>) {
145        if !self.enabled(record.metadata()) {
146            return;
147        }
148
149        // The logger being uninitialized is impossible, as that would mean it
150        // wouldn't have been registered with the log library.
151        // Log format/detail depends on the level.
152        let record_args = record.args();
153        match record.metadata().level() {
154            log::Level::Error | log::Level::Warn => {
155                let level_str = record.metadata().level().as_str();
156                lb_write(format_args!("{level_str}: {record_args}\r\n"));
157            }
158
159            log::Level::Info => {
160                lb_write(format_args!("{record_args}\r\n"));
161            }
162
163            log::Level::Debug | log::Level::Trace => {
164                let level_str = record.metadata().level().as_str();
165                let tgt = record.metadata().target();
166                lb_write(format_args!("[{tgt}] {level_str} {record_args}\r\n"));
167            }
168        };
169    }
170
171    fn flush(&self) {}
172}

174static BUFFER_LOGGER: BufferLogger = BufferLogger;

176/// Registers the logger to collect startup messages until
177/// [`switch_to_log_buffer()`]. Earlier messages are discarded.
178///
179/// # Panics
180///
181/// Panics if a logger is already registered.
182pub fn install_early_logger() {
183    // As described in the `log` crate, this function can be called only once
184    // and successive calls will return an error. Panic on that error.
185    // Essentially, it handles initialization in the same way as `immut_after_init`,
186    // so we can avoid using it. Therefore, the logger will be immutable after initialization.
187    log::set_logger(&BUFFER_LOGGER).expect("Failed to install the BufferLogger");
188
189    // Log levels are to be configured via the log's library feature configuration.
190    log::set_max_level(log::LevelFilter::Trace);
191    log::info!("COCONUT Secure Virtual Machine Service Module Version {COCONUT_VERSION}");
192}

194/// Writes `bytes` to the log file and, with the `enable-console-log` feature,
195/// to the console. This bypasses the line buffer, so the caller is responsible
196/// for the message prefixes and the line breaks.
197fn write_log_sinks(bytes: &[u8]) {
198    let _ = log_write(bytes).map_err(print_warning);
199
200    #[cfg(feature = "enable-console-log")]
201    console_write(bytes);
202}

204/// Activates the regular log buffer and transfers the early logs.
205///
206/// Requires the file system, log file, idle task, and console to be initialized.
207///
208/// # Panics
209///
210/// Panics if called more than once.
211pub fn switch_to_log_buffer() {
212    let logbuf = SpinLock::new(LogBuffer::new());
213    LOGGER
214        .init(logbuf)
215        .expect("Failed to initialize log buffer");
216
217    // Initializing the logger above routes new messages to the regular log
218    // buffer, so the writes below cannot take this lock again even though
219    // they can emit log messages themselves.
220    let early = EARLY_LOG.lock();
221
222    // The early log contents already carry prefixes and line breaks, so they
223    // go to the log file directly.
224    write_log_sinks(early.contents());
225    if early.truncated {
226        write_log_sinks(EARLY_LOG_TRUNCATED.as_bytes());
227    }
228}
