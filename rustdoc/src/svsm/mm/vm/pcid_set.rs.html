7//! Shared set of PCIDs that may cache translations for a VMR.
8//!
9//! Threads can share VMRs while each having its own page-table root and PCID.
10//! Unmaps through those VMRs must shoot down every PCID that could still hold
11//! the old translations.
12
13use crate::address::VirtAddr;
14use crate::cpu::{flush_tlb_global_sync_range, flush_tlb_pcid_sync_range};
15use crate::locking::SpinLock;
16use crate::types::PageSize;
17use crate::utils::MemoryRegion;
18use core::sync::atomic::{AtomicUsize, Ordering};
19
20extern crate alloc;
21use alloc::sync::Arc;
22use alloc::vec::Vec;
23
24/// PCIDs of page-table roots that may contain translations for a shared VMR.
25#[derive(Debug, Default)]
26pub struct PcidSet {
27    /// Allocated PCIDs (unique per page-table root).
28    ids: SpinLock<Vec<u16>>,
29    /// Tasks that fell back to reserved PCID 0 when the pool was exhausted.
30    overflow: AtomicUsize,
31}
32
33impl PcidSet {
34    pub fn new() -> Arc<Self> {
35        Arc::new(Self::default())
36    }
37
38    pub fn insert(&self, pcid: u16) {
39        if pcid == 0 {
40            self.overflow.fetch_add(1, Ordering::Relaxed);
41            return;
42        }
43        let mut ids = self.ids.lock();
44        if !ids.contains(&pcid) {
45            ids.push(pcid);
46        }
47    }
48
49    pub fn remove(&self, pcid: u16) {
50        if pcid == 0 {
51            // Underflow means an unmatched remove; ignore it.
52            loop {
53                let n = self.overflow.load(Ordering::Relaxed);
54                if n == 0 {
55                    return;
56                }
57                if self
58                    .overflow
59                    .compare_exchange_weak(n, n - 1, Ordering::Relaxed, Ordering::Relaxed)
60                    .is_ok()
61                {
62                    return;
63                }
64            }
65        }
66        self.ids.lock().retain(|&id| id != pcid);
67    }
68
69    /// Invalidate `region` for every registered PCID, or globally if none.
70    pub fn flush_range(&self, region: MemoryRegion<VirtAddr>, pgsize: PageSize) {
71        let ids = self.ids.lock().clone();
72        let overflow = self.overflow.load(Ordering::Relaxed);
73        if ids.is_empty() && overflow == 0 {
74            flush_tlb_global_sync_range(region, pgsize);
75            return;
76        }
77        for pcid in ids {
78            flush_tlb_pcid_sync_range(pcid, region, pgsize);
79        }
80        if overflow > 0 {
81            flush_tlb_pcid_sync_range(0, region, pgsize);
82        }
83    }
84}
