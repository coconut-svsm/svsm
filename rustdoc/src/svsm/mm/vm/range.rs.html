31/// Granularity of ranges mapped by [`struct VMR`]. The mapped region of a
32/// [`struct VMR`] is always a multiple of this constant.
33/// One [`VMR_GRANULE`] covers one top-level page-table entry on x86-64 with
34/// 4-level paging.
35pub const VMR_GRANULE: usize = PAGE_SIZE * 512 * 512 * 512;
36
37/// Virtual Memory Region
38///
39/// This struct manages the mappings in a region of the virtual address space.
40/// The region size is a multiple of 512GiB so that every region will fully
41/// allocate one or more top-level page-table entries on x86-64. For the same
42/// reason the start address must also be aligned to 512GB.
43#[derive(Debug)]
44pub struct VMR {
45    /// Start address of this range as virtual PFN (VirtAddr >> PAGE_SHIFT).
46    /// Virtual address must be aligned to [`VMR_GRANULE`] (512GB on x86-64).
47    start_pfn: usize,
48
49    /// End address of this range as virtual PFN (VirtAddr >> PAGE_SHIFT)
50    /// Virtual address must be aligned to [`VMR_GRANULE`] (512GB on x86-64).
51    end_pfn: usize,
52
53    /// RBTree containing all [`struct VMM`] instances with valid mappings in
54    /// the covered virtual address region. The [`struct VMM`]s are sorted by
55    /// their start address and stored in an RBTree for faster lookup.
56    tree: RWLock<RBTree<VMMAdapter>>,
57
58    /// [`struct PageTableParts`] needed to map this VMR into a page-table.
59    /// There is one [`struct PageTablePart`] per [`VMR_GRANULE`] covered by
60    /// the region.
61    pgtbl_parts: RWLock<Vec<PageTablePart>>,
62
63    /// [`PTEntryFlags`] global to all mappings in this region. This is a
64    /// combination of [`PTEntryFlags::GLOBAL`] and [`PTEntryFlags::USER`].
65    pt_flags: PTEntryFlags,
66
67    /// Indicates that this [`struct VMR`] is visible only on a single CPU
68    /// and therefore TLB flushes do not require broadcast.
69    per_cpu: bool,
70
71    /// PCIDs of page-table roots that may cache translations for this VMR.
72    /// `None` for global/shared VMRs. Task VMRs share one set across threads:
73    /// each thread has its own root/PCID, so unmaps must shoot down every
74    /// member rather than a single id or the current CPU's PCID.
75    pcid_set: Option<Arc<PcidSet>>,
76}

78impl VMR {
79    /// Creates a new [`struct VMR`]
80    ///
81    /// # Arguments
82    ///
83    /// * `start` - Virtual start address for the memory region. Must be aligned to [`VMR_GRANULE`]
84    /// * `end` - Virtual end address (non-inclusive) for the memory region.
85    ///   Must be bigger than `start` and aligned to [`VMR_GRANULE`].
86    /// * `flags` - Global [`PTEntryFlags`] to use for this [`struct VMR`].
87    ///
88    /// # Returns
89    ///
90    /// A new instance of [`struct VMR`].
91    pub fn new(start: VirtAddr, end: VirtAddr, flags: PTEntryFlags) -> Result<Self, SvsmError> {
92        if start >= end || !start.is_aligned(VMR_GRANULE) || !end.is_aligned(VMR_GRANULE) {
93            log::warn!("Attempted to create an invalid VMR {start:#018x}-{start:#018x}");
94            return Err(SvsmError::Mem);
95        }
96        // Global and User are per VMR flags
97        Ok(Self {
98            start_pfn: start.pfn(),
99            end_pfn: end.pfn(),
100            tree: RWLock::new(RBTree::new(VMMAdapter::new())),
101            pgtbl_parts: RWLock::new(Vec::new()),
102            pt_flags: flags,
103            per_cpu: false,
104            pcid_set: None,
105        })
106    }

124    /// Allocated all [`PageTablePart`]s needed to map this region
125    ///
126    /// # Returns
127    ///
128    /// `Ok(())` on success, Err(SvsmError::Mem) on allocation error
129    fn alloc_page_tables(&self, lazy: bool) -> Result<(), SvsmError> {
130        let vregion = self.virt_range();
131
132        let first_idx = vregion.start().to_pgtbl_idx::<3>();
133        let start = virt_from_idx(first_idx);
134        let last_idx = (vregion.end() - 1).to_pgtbl_idx::<3>();
135        let count = last_idx + 1 - first_idx;
136        let mut vec = self.pgtbl_parts.lock_write();
137
138        for idx in 0..count {
139            let mut part = PageTablePart::new(start + (idx * VMR_GRANULE));
140            if !lazy {
141                part.alloc();
142            }
143            vec.push(part);
144        }
145
146        Ok(())
147    }

149    /// Populate [`PageTablePart`]s of the [`VMR`] into a page-table
150    ///
151    /// # Arguments
152    ///
153    /// * `pgtbl` - A [`PageTable`] pointing to the target page-table
154    pub fn populate(&self, pgtbl: &mut PageTable) {
155        let parts = self.pgtbl_parts.lock_read();
156
157        for part in parts.iter() {
158            pgtbl.populate_pgtbl_part(part);
159        }
160    }
161
162    fn populate_addr(&self, pgtbl: &mut PageTable, vaddr: VirtAddr) -> Result<(), SvsmError> {
163        let vregion = self.virt_range();
164        if !vregion.contains(vaddr) {
165            return Err(SvsmError::Mem);
166        }
167
168        let idx = vaddr.to_pgtbl_idx::<3>() - vregion.start().to_pgtbl_idx::<3>();
169        let parts = self.pgtbl_parts.lock_read();
170        if !pgtbl.populate_pgtbl_part(&parts[idx]) {
171            return Err(SvsmError::Mem);
172        }
173        Ok(())
174    }

206    /// Returns the virtual start and end addresses for this region

208    /// # Returns

211    pub fn virt_range(&self) -> MemoryRegion<VirtAddr> {

218    /// Map a [`VMM`] into the [`PageTablePart`]s of this region
219    ///
220    /// # Arguments
221    ///
222    /// - `vmm` - Reference to a [`VMM`] instance to map into the page-table
223    ///
224    /// # Returns
225    ///
226    /// `Ok(())` on success, Err(SvsmError::Mem) on allocation error
227    fn map_vmm(&self, vmm: &VMM) -> Result<(), SvsmError> {
228        let rstart = self.virt_range().start();
229        let (vmm_start, vmm_end) = vmm.range();
230        let mut pgtbl_parts = self.pgtbl_parts.lock_write();
231        let mapping = vmm.get_mapping();
232        let mut offset: usize = 0;
233        let page_size = mapping.page_size();
234        let shared = mapping.shared();
235
236        // Exit early if the mapping has no data.
237        if !mapping.has_data() {
238            return Ok(());
239        }
240
241        while vmm_start + offset < vmm_end {
242            let idx = PageTable::index::<3>(VirtAddr::from(vmm_start - rstart));
243            if let Some(paddr) = mapping.map(offset) {
244                let pt_flags = self.pt_flags | mapping.pt_flags(offset) | PTEntryFlags::PRESENT;
245                match page_size {
246                    PageSize::Regular => {
247                        pgtbl_parts[idx].map_4k(vmm_start + offset, paddr, pt_flags, shared)?
248                    }
249                    PageSize::Huge => {
250                        pgtbl_parts[idx].map_2m(vmm_start + offset, paddr, pt_flags, shared)?
251                    }
252                }
253            }
254            offset += usize::from(page_size);
255        }
256
257        Ok(())
258    }

260    /// Unmap a [`VMM`] from the [`PageTablePart`]s of this region
261    ///
262    /// # Arguments
263    ///
264    /// - `vmm` - Reference to a [`VMM`] instance to unmap from the page-table
265    fn unmap_vmm(&self, vmm: &VMM) {
266        let rstart = self.virt_range().start();
267        let (vmm_start, vmm_end) = vmm.range();
268        let mut pgtbl_parts = self.pgtbl_parts.lock_write();
269        let mapping = vmm.get_mapping();
270        let page_size = mapping.page_size();
271        let mut offset: usize = 0;
272
273        while vmm_start + offset < vmm_end {
274            let idx = PageTable::index::<3>(VirtAddr::from(vmm_start - rstart));
275            let result = match page_size {
276                PageSize::Regular => pgtbl_parts[idx].unmap_4k(vmm_start + offset),
277                PageSize::Huge => pgtbl_parts[idx].unmap_2m(vmm_start + offset),
278            };
279
280            if result.is_some() {
281                mapping.unmap(offset);
282            }
283
284            offset += usize::from(page_size);
285        }
286    }

288    fn do_insert(
289        &self,
290        mapping: Mapping,
291        start_pfn: usize,
292        cursor: &mut CursorMut<'_, VMMAdapter>,
293    ) -> Result<(), SvsmError> {
294        let vmm = Box::new(VMM::new(start_pfn, mapping));
295        if let Err(e) = self.map_vmm(&vmm) {
296            self.unmap_vmm(&vmm);
297            Err(e)
298        } else {
299            cursor.insert_before(vmm);
300            Ok(())
301        }
302    }

304    /// Inserts [`VMM`] at a specified virtual base address. This method
305    /// checks that the [`VMM`] does not overlap with any other region.
306    ///
307    /// # Arguments
308    ///
309    /// * `vaddr` - Virtual base address to map the [`VMM`] at
310    /// * `mapping` - `Arc` pointer to the VMM to insert
311    ///
312    /// # Returns
313    ///
314    /// Base address where the [`VMM`] was inserted on success or SvsmError::Mem on error
315    pub fn insert_at(&self, vaddr: VirtAddr, mapping: Mapping) -> Result<VirtAddr, SvsmError> {
316        // mapping-size needs to be page-aligned
317        let size = mapping.mapping_size() >> PAGE_SHIFT;
318        let start_pfn = vaddr.pfn();
319        let mut tree = self.tree.lock_write();
320        let mut cursor = tree.upper_bound_mut(Bound::Included(&start_pfn));
321        let mut start = self.start_pfn;
322        let mut end = self.end_pfn;
323
324        if cursor.is_null() {
325            cursor = tree.front_mut();
326        } else {
327            let (_, node_end) = cursor.get().unwrap().range_pfn();
328            start = node_end;
329            cursor.move_next();
330        }
331
332        if let Some(node) = cursor.get() {
333            let (node_start, _) = node.range_pfn();
334            end = node_start;
335        }
336
337        let end_pfn = start_pfn + size;
338
339        if start_pfn >= start && end_pfn <= end {
340            self.do_insert(mapping, start_pfn, &mut cursor)?;
341            Ok(vaddr)
342        } else {
343            Err(SvsmError::Mem)
344        }
345    }

347    /// Inserts [`VMM`] with the specified alignment. This method walks the
348    /// RBTree to search for a suitable region.
349    ///
350    /// # Arguments
351    ///
352    /// * `mapping` - `Arc` pointer to the VMM to insert
353    /// * `align` - Alignment to use for tha mapping
354    ///
355    /// # Returns
356    ///
357    /// Base address where the [`VMM`] was inserted on success or SvsmError::Mem on error
358    pub fn insert_aligned(
359        &self,
360        hint: VirtAddr,
361        mapping: Mapping,
362        align: usize,
363    ) -> Result<VirtAddr, SvsmError> {
364        assert!(align.is_power_of_two());
365
366        let size = mapping
367            .mapping_size()
368            .checked_next_power_of_two()
369            .unwrap_or(0)
370            >> PAGE_SHIFT;
371        let align = align >> PAGE_SHIFT;
372
373        let start_pfn = max(self.start_pfn, hint.pfn());
374
375        let mut start = align_up(start_pfn, align);
376        let mut end = start;
377
378        if size == 0 || start_pfn >= self.end_pfn {
379            return Err(SvsmError::Mem);
380        }
381
382        let mut tree = self.tree.lock_write();
383        let mut cursor = tree.upper_bound_mut(Bound::Included(&start_pfn));
384        if cursor.is_null() {
385            cursor = tree.front_mut();
386        }
387
388        while let Some(node) = cursor.get() {
389            let (node_start, node_end) = node.range_pfn();
390            end = node_start;
391            if end > start && end - start >= size {
392                break;
393            }
394
395            start = max(start, align_up(node_end, align));
396            cursor.move_next();
397        }
398
399        if cursor.is_null() {
400            end = align_down(self.end_pfn, align);
401        }
402
403        if end > start && end - start >= size {
404            self.do_insert(mapping, start, &mut cursor)?;
405            Ok(VirtAddr::from(start << PAGE_SHIFT))
406        } else {
407            Err(SvsmError::Mem)
408        }
409    }

411    /// Inserts [`VMM`] into the virtual memory region. This method takes the
412    /// next power-of-two larger of the mapping size and uses that as the
413    /// alignment for the mappings base address. The search for the base
414    /// address starts at `addr`. With that it calls [`VMR::insert_aligned`].
415    ///
416    /// # Arguments
417    ///
418    /// * `addr` - The virtual address at which the search for a mapping area
419    ///   starts
420    /// * `mapping` - `Arc` pointer to the VMM to insert
421    ///
422    /// # Returns
423    ///
424    /// Base address where the [`VMM`] was inserted on success or SvsmError::Mem on error
425    pub fn insert_hint(&self, addr: VirtAddr, mapping: Mapping) -> Result<VirtAddr, SvsmError> {
426        let align = mapping.mapping_size().next_power_of_two();
427        self.insert_aligned(addr, mapping, align)
428    }

444    /// Removes the mapping from a given base address from the RBTree
445    ///
446    /// # Arguments
447    ///
448    /// * `base` - Virtual base address of the [`VMM`] to remove
449    ///
450    /// # Returns
451    ///
452    /// The removed mapping on success, SvsmError::Mem on error
453    pub fn remove(&self, base: VirtAddr) -> Result<Box<VMM>, SvsmError> {
454        let mut tree = self.tree.lock_write();
455        let addr = base.pfn();
456
457        let mut cursor = tree.find_mut(&addr);
458        let node = cursor.remove().ok_or(SvsmError::Mem)?;
459        self.unmap_vmm(&node);
460
461        let range = node.range();
462        let region = MemoryRegion::from_addresses(range.0, range.1);
463        let pgsize = node.get_mapping().page_size();
464
465        if self.per_cpu {
466            flush_tlb_global_percpu_range(region, pgsize);
467        } else if let Some(ref pcid_set) = self.pcid_set {
468            // The page-table update above must complete before this
469            // shootdown. The synchronous flush must complete before the
470            // unmapped page can be freed or reassigned.
471            pcid_set.flush_range(region, pgsize);
472        } else {
473            flush_tlb_global_sync_range(region, pgsize);
474        }
475
476        Ok(node)
477    }

493    /// Handle a page fault for an address corresponding to this VMR.
494    ///
495    /// The fault is first handled by attemping to populate the provided page table
496    /// with the page table parts corresponding to the faulting address. If that
497    /// does not solve the fault, notify the backing mapping that a page fault has
498    /// occurred.
499    ///
500    /// This should be called from the page fault handler. The mappings within this
501    /// virtual memory region are examined and if they overlap with the page fault
502    /// address then [`VirtualMapping::handle_page_fault`] is called to handle the
503    /// page fault within that range.
504    ///
505    /// [`VirtualMapping::handle_page_fault`]: super::mapping::api::VirtualMapping::handle_page_fault
506    ///
507    /// # Arguments
508    ///
509    /// * `pgtable`: The page table to update with the faulted-in mapping, if applicable.
510    /// * `vaddr` - Virtual memory address that was the subject of the page fault
511    /// * 'write' - 'true' if a write was attempted. 'false' if a read was attempted.
512    ///
513    /// # Returns
514    ///
515    /// '()' if the page fault was successfully handled.
516    ///
517    /// 'SvsmError::Mem' if the page fault should propogate to the next handler.
518    pub fn handle_page_fault(
519        &self,
520        pgtable: &mut PageTable,
521        vaddr: VirtAddr,
522        write: bool,
523    ) -> Result<(), SvsmError> {
524        // Check first if the fault is solved by populating the page table
525        if let Ok(()) = self.populate_addr(pgtable, vaddr) {
526            return Ok(());
527        }
528
529        // Get the mapping that contains the faulting address and check if the
530        // fault happened on a mapped part of the range.
531        let tree = self.tree.lock_read();
532        let pfn = vaddr.pfn();
533        let cursor = tree.upper_bound(Bound::Included(&pfn));
534        let node = cursor.get().ok_or(SvsmError::Mem)?;
535        let (start, end) = node.range();
536        if vaddr < start || vaddr >= end {
537            return Err(SvsmError::Mem);
538        }
539
540        let off = vaddr - start;
541        node.get_mapping().handle_page_fault(self, off, write)?;
542        Ok(())
543    }
544}

546/// A mapping in a [`VMR`], holding a reference `V` to that `VMR`.
547/// The mapping is torn down on drop.
548#[derive(Debug)]
549pub struct VMRMapping<V: Borrow<VMR>> {
550    vmr: V,
551    va: VirtAddr,
552}
553
554impl<V: Borrow<VMR>> VMRMapping<V> {
555    pub fn new(vmr: V, mapping: Mapping) -> Result<Self, SvsmError> {
556        let va = vmr.borrow().insert(mapping)?;
557        Ok(Self { vmr, va })
558    }
559
560    pub fn new_at(vmr: V, addr: VirtAddr, mapping: Mapping) -> Result<Self, SvsmError> {
561        let va = vmr.borrow().insert_at(addr, mapping)?;
562        Ok(Self { vmr, va })
563    }
564
565    pub fn new_hint(vmr: V, addr: VirtAddr, mapping: Mapping) -> Result<Self, SvsmError> {
566        let va = vmr.borrow().insert_hint(addr, mapping)?;
567        Ok(Self { vmr, va })
568    }
569
570    pub fn leak(self) -> VirtAddr {
571        let md = ManuallyDrop::new(self);
572        md.va
573    }
574
575    pub fn virt_addr(&self) -> VirtAddr {
576        self.va
577    }
578}
579
580impl<V: Borrow<VMR>> Deref for VMRMapping<V> {
581    type Target = VirtAddr;
582
583    fn deref(&self) -> &VirtAddr {
584        &self.va
585    }
586}
587
588impl<V: Borrow<VMR>> Drop for VMRMapping<V> {
589    fn drop(&mut self) {
590        self.vmr
591            .borrow()
592            .remove(self.va)
593            .expect("Error removing VRMapping virtual memory range");
594    }
