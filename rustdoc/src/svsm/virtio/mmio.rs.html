20#[derive(Debug)]
21pub struct MmioSlot {
22    pub transport: MmioTransport<'static>,
23    // Fields are ordered so that `mmio_range` is dropped last.
24    // The MmioTransport destructor resets the device via MMIO writes, and
25    // dropping `mmio_range` destroys the mapping, so `mmio_range` must be
26    // dropped last.
27    // This drop-order behavior is stable.
28    // See https://doc.rust-lang.org/reference/destructors.html
29    //
30    // TODO: The destruction order should be expressed via code rather than
31    // relying on field ordering. This should be addressed when moving to
32    // the upstream virtio-drivers crate.
33    pub mmio_range: GlobalRangeGuard,
34}

36#[derive(Debug, Default)]
37pub struct MmioSlots {
38    slots: Vec<MmioSlot>,
39}

41#[derive(Debug)]
42struct MmioRegion {
43    address: usize,
44    size: usize,
45}
46
47/// Parses the device tree to extract base addresses and size of virtio-MMIO
48/// devices. Devices without a `plane` property are always included; devices
49/// with a `plane` property are included only if it matches [`SVSM_VMPL`].
50fn get_virtio_mmio_addresses(device_tree: Fdt<'_>) -> Vec<MmioRegion> {
51    device_tree
52        .all_nodes()
53        .filter(|dev| {
54            dev.compatible()
55                .is_some_and(|c| c.all().any(|c| c == "virtio,mmio"))
56        })
57        .filter(|dev| {
58            let Some(prop) = dev.property("plane") else {
59                return true;
60            };
61            let Some(plane) = prop.as_usize() else {
62                log::warn!("virtio-mmio: malformed 'plane' property, skipping");
63                return false;
64            };
65            plane == SVSM_VMPL
66        })
67        .filter_map(|element| {
68            let reg_value = element.reg()?.next()?;
69
70            Some(MmioRegion {
71                address: reg_value.starting_address as usize,
72                size: reg_value.size?,
73            })
74        })
75        .collect()
76}

78/// Probes and enumerates all virtio-MMIO devices available in the system.
79///
80/// This function parses the device tree to discover virtio-MMIO device
81/// addresses and maps their MMIO regions.
82///
83/// # Usage
84///
85/// This function is typically called once during early system initialization
86/// to discover all available virtio devices. Each slot in the returned collection
87/// should be consumed exactly once by calling [`MmioSlots::pop_slot`], as each
88/// slot represents exclusive ownership of a device's MMIO region.
89///
90/// # Returns
91///
92/// Returns an [`MmioSlots`] collection containing all discovered virtio-MMIO devices.
93/// Returns an empty collection if no devices are found or if the device tree
94/// is unavailable.
95pub fn probe_mmio_slots(boot_params: &BootParams<'_>) -> MmioSlots {
96    let Ok(parsed_dt) = Fdt::new(boot_params.get_device_tree()) else {
97        log::warn!("MmioSlots: Failed to parse device tree");
98        return MmioSlots::default();
99    };
100
101    let dev = get_virtio_mmio_addresses(parsed_dt);
102    let mut slots = Vec::with_capacity(dev.len());
103
104    for slot in dev {
105        let phys_addr = PhysAddr::from(slot.address);
106        let page_base = phys_addr.page_align();
107        let page_offset = phys_addr.page_offset();
108
109        if !phys_addr.is_aligned_to::<VirtIOHeader>() {
110            log::warn!("MmioSlots: MMIO device at {phys_addr:x} is not properly aligned");
111            continue;
112        }
113
114        if phys_addr.crosses_page(slot.size) {
115            log::warn!("MmioSlots: MMIO device region at {phys_addr:x} crosses a page boundary");
116            continue;
117        }
118
119        // If multiple devices reside in the same page, each gets its own
120        // mapping, which is slightly wasteful but avoids shared-ownership
121        // complexity.
122        let Ok(mem) = map_global_range_4k_shared(page_base, slot.size, PTEntryFlags::data()) else {
123            log::warn!("MmioSlots: Failed to map MMIO region at {:x}", slot.address);
124            continue;
125        };
126
127        // Not expected to fail, because mem exists.
128        let header = NonNull::new((mem.addr() + page_offset).as_mut_ptr()).unwrap();
129
130        // SAFETY: The address is valid, mapped by `map_global_range_4k_shared`, and verified
131        // to be VirtIOHeader-aligned by the guard above.
132        // The memory region has the same lifetime of the MmioSlot structure which will be consumed by the driver.
133        let Ok(transport) = (unsafe { MmioTransport::new(header, slot.size) }) else {
134            // Currently QEMU advertises _all_ slots, regardless they are empty or not.
135            log::debug!("MmioSlots: {:x} empty", slot.address);
136            continue;
137        };
138
139        log::info!(
140            "MmioSlots: Found {:?} at {:x}",
141            transport.device_type(),
142            slot.address
143        );
144
145        let slot_type = MmioSlot {
146            transport,
147            mmio_range: mem,
148        };
149
150        slots.push(slot_type);
151    }
152
153    MmioSlots { slots }
154}
155
156impl MmioSlots {
157    /// Retrieves and removes the first available MMIO slot matching the specified device type.
158    ///
159    /// This method consumes the slot. Once retrieved, the slot is removed
160    /// from the array and cannot be obtained again. This ensures each
161    /// virtio device is initialized exactly once and forbids driver unloading.
162    ///
163    /// # Arguments
164    ///
165    /// * `virtio_type` - The virtio device type to search for
166    ///
167    /// # Returns
168    ///
169    /// * `Some(MmioSlot)` - The first matching slot
170    /// * `None` - No slot matching the requested device type was found
171    pub fn pop_slot(&mut self, virtio_type: DeviceType) -> Option<MmioSlot> {
172        let pos = self
173            .slots
174            .iter()
175            .position(|slot| slot.transport.device_type() == virtio_type)?;
176        Some(self.slots.remove(pos))
177    }
178}
