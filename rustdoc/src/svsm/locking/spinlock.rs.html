16/// A lock guard obtained from a [`SpinLock`]. This lock guard
17/// provides exclusive access to the data protected by a [`SpinLock`],
18/// ensuring that the lock is released when it goes out of scope.
19///
20/// # Examples
21///
22/// ```
23/// use svsm::locking::SpinLock;
24///
25/// let data = 42;
26/// let spin_lock = SpinLock::new(data);
27///
28/// {
29///     let mut guard = spin_lock.lock();
30///     *guard += 1; // Modify the protected data.
31/// }; // Lock is automatically released when `guard` goes out of scope.
32/// ```
33#[derive(Debug)]
34#[must_use = "if unused the SpinLock will immediately unlock"]
35pub struct RawLockGuard<'a, T, I> {
36    holder: &'a AtomicU64,
37    /// Pointer to the protected data. This relaxes the borrow checker
38    /// when implementing `map()` and related methods, and prevents
39    /// introducing LLVM `noalias` violations, according to a comment
40    /// in the equivalent guard structure for RwLock in the standard
41    /// library.
42    data: NonNull<T>,
43    _variance: PhantomData<&'a mut T>,
44    irq_state: I,
45}
46
47impl<'a, T, I: IrqLocking> RawLockGuard<'a, T, I> {
48    pub fn map<U, F>(orig: Self, f: F) -> RawLockGuard<'a, U, I>
49    where
50        F: FnOnce(&mut T) -> &mut U,
51    {
52        let mut orig = ManuallyDrop::new(orig);
53        let holder = orig.holder;
54        // Move the original IRQ state out of drop.
55        // SAFETY: we are really reading from a reference, so the source
56        // pointer is safe. The original guard is behind `ManuallyDrop`,
57        // so only the copy we just make will invoke drop.
58        let irq_state = unsafe { core::ptr::read(&raw const orig.irq_state) };
59        let value = f(&mut *orig);
60        RawLockGuard {
61            holder,
62            data: NonNull::from(value),
63            _variance: PhantomData,
64            irq_state,
65        }
66    }
67}

69// SAFETY: RawLockGuard does not automatically implement Sync because it
70// contains a `NonNull`, which is guarded by the lock's behavior.
71unsafe impl<T: Sync, I: Sync> Sync for RawLockGuard<'_, T, I> {}
72
73// SAFETY: RawLockGuard does not automatically implement Send because it
74// contains a `NonNull`, which is guarded by the lock's behavior.
75unsafe impl<T: Send, I: Send> Send for RawLockGuard<'_, T, I> {}

77/// Implements the behavior of the [`LockGuard`] when it is dropped
78impl<T, I> Drop for RawLockGuard<'_, T, I> {
79    /// Automatically releases the lock when the guard is dropped
80    fn drop(&mut self) {
81        self.holder.fetch_add(1, Ordering::Release);
82    }
83}
84
85/// Implements the behavior of dereferencing the [`LockGuard`] to
86/// access the protected data.
87impl<T, I> Deref for RawLockGuard<'_, T, I> {
88    type Target = T;
89    /// Provides read-only access to the protected data
90    fn deref(&self) -> &T {
91        // SAFETY: a spinlock guard guarantees exclusive access
92        unsafe { self.data.as_ref() }
93    }
94}

96/// Implements the behavior of dereferencing the [`LockGuard`] to
97/// access the protected data in a mutable way.
98impl<T, I> DerefMut for RawLockGuard<'_, T, I> {
99    /// Provides mutable access to the protected data
100    fn deref_mut(&mut self) -> &mut T {
101        // SAFETY: a spinlock guard guarantees exclusive access
102        unsafe { self.data.as_mut() }
103    }
104}

110/// A simple ticket-spinlock implementation for protecting concurrent data
111/// access.
112///
113/// Two variants are derived from this implementation:
114///
115///  * [`SpinLock`] for general use. This implementation is not safe for use in
116///    IRQ handlers.
117///  * [`SpinLockIrqSafe`] for protecting data that is accessed in IRQ context.
118///
119/// # Examples
120///
121/// ```
122/// use svsm::locking::SpinLock;
123///
124/// let data = 42;
125/// let spin_lock = SpinLock::new(data);
126///
127/// // Acquire the lock and modify the protected data.
128/// {
129///     let mut guard = spin_lock.lock();
130///     *guard += 1;
131/// }; // Lock is automatically released when `guard` goes out of scope.
132///
133/// // Try to acquire the lock without blocking
134/// if let Some(mut guard) = spin_lock.try_lock() {
135///     *guard += 2;
136/// };
137/// ```
138#[derive(Debug, Default)]
139pub struct RawSpinLock<T, I> {
140    /// This atomic counter is incremented each time a thread attempts to
141    /// acquire the lock. It helps to determine the order in which threads
142    /// acquire the lock.
143    current: AtomicU64,
144    /// This counter represents the thread that currently holds the lock
145    /// and has access to the protected data.
146    holder: AtomicU64,
147    /// This `UnsafeCell` is used to provide interior mutability of the
148    /// protected data. That is, it allows the data to be accessed/modified
149    /// while enforcing the locking mechanism.
150    data: UnsafeCell<T>,
151    /// Use generic type I in the struct without consuming space.
152    phantom: PhantomData<fn(I)>,
153}
154
155// SAFETY: A well-formed lock is always `Send`.
156unsafe impl<T, I> Send for RawSpinLock<T, I> {}
157// SAFETY: A well-formed lock is always `Sync`.
158unsafe impl<T, I> Sync for RawSpinLock<T, I> {}
159
160/// A lock can only be formed if the type it protects is `Send`, since the
161/// contents of the lock will be sent to different threads.
162impl<T: Send, I: IrqLocking> RawSpinLock<T, I> {
163    /// Creates a new SpinLock instance with the specified initial data.
164    ///
165    /// # Examples
166    ///
167    /// ```
168    /// use svsm::locking::SpinLock;
169    ///
170    /// let data = 42;
171    /// let spin_lock = SpinLock::new(data);
172    /// ```
173    pub const fn new(data: T) -> Self {
174        Self {
175            current: AtomicU64::new(0),
176            holder: AtomicU64::new(0),
177            data: UnsafeCell::new(data),
178            phantom: PhantomData,
179        }
180    }
181
182    /// Acquires the lock, providing access to the protected data.
183    ///
184    /// # Examples
185    ///
186    /// ```
187    /// use svsm::locking::SpinLock;
188    ///
189    /// let spin_lock = SpinLock::new(42);
190    ///
191    /// // Acquire the lock and modify the protected data.
192    /// {
193    ///     let mut guard = spin_lock.lock();
194    ///     *guard += 1;
195    /// }; // Lock is automatically released when `guard` goes out of scope.
196    /// ```
197    pub fn lock(&self) -> RawLockGuard<'_, T, I> {
198        let irq_state = I::acquire_lock();
199
200        let ticket = self.current.fetch_add(1, Ordering::Relaxed);
201        loop {
202            let h = self.holder.load(Ordering::Acquire);
203            if h == ticket {
204                break;
205            }
206            core::hint::spin_loop();
207        }
208        RawLockGuard {
209            holder: &self.holder,
210            // SAFETY: the UnsafeCell is initialized on construction, so the
211            // pointer can never be NULL
212            data: unsafe { NonNull::new_unchecked(self.data.get()) },
213            _variance: PhantomData,
214            irq_state,
215        }
216    }
217
218    /// Execute function F while holding the lock.
219    ///
220    /// # Examples
221    ///
222    /// ```
223    /// use svsm::locking::SpinLock;
224    ///
225    /// let spin_lock = SpinLock::new(42);
226    ///
227    /// // Do some actions while holding the lock.
228    /// // Lock is automatically taken and released.
229    /// spin_lock.locked_do(|s| {
230    ///     *s += 1;
231    /// });
232    /// ```
233    pub fn locked_do<R, F: FnMut(&mut T) -> R>(&self, mut f: F) -> R {
234        let mut l = self.lock();
235        f(&mut (*l))
236    }
237
238    /// This method tries to acquire the lock without blocking. If the
239    /// lock is not available, it returns `None`. If the lock is
240    /// successfully acquired, it returns a [`LockGuard`] that automatically
241    /// releases the lock when it goes out of scope.
242    pub fn try_lock(&self) -> Option<RawLockGuard<'_, T, I>> {
243        let irq_state = I::acquire_lock();
244
245        let current = self.current.load(Ordering::Relaxed);
246        let holder = self.holder.load(Ordering::Acquire);
247
248        if current == holder {
249            let result = self.current.compare_exchange(
250                current,
251                current + 1,
252                Ordering::Acquire,
253                Ordering::Relaxed,
254            );
255            if result.is_ok() {
256                return Some(RawLockGuard {
257                    holder: &self.holder,
258                    // SAFETY: the UnsafeCell is initialized on construction, so the
259                    // pointer can never be NULL
260                    data: unsafe { NonNull::new_unchecked(self.data.get()) },
261                    _variance: PhantomData,
262                    irq_state,
263                });
264            }
265        }
266
267        None
268    }
269
270    /// Returns a mutable reference to the underlying data.
271    ///
272    /// Since this call borrows the `RawSpinLock` mutably, no actual locking needs to take place --
273    /// the mutable borrow statically guarantees no new locks can be acquired while this reference
274    /// exists.
275    pub fn get_mut(&mut self) -> &mut T {
276        // SAFETY: the returned reference carries an exclusive borrow on self,
277        // thereby establishing exclusive access.
278        unsafe { &mut *self.data.get() }
279    }
280}

282impl<T: Send, I: IrqLocking> From<T> for RawSpinLock<T, I> {
283    fn from(value: T) -> Self {
284        Self::new(value)
285    }
286}

300    #[test]
301    fn test_spin_lock() {
302        let spin_lock = SpinLock::new(0);
303
304        let mut guard = spin_lock.lock();
305        *guard += 1;
306
307        // Ensure the locked data is updated.
308        assert_eq!(*guard, 1);
309
310        // Try to lock again; it should fail and return None.
311        let try_lock_result = spin_lock.try_lock();
312        assert!(try_lock_result.is_none());
313    }

315    #[test]
316    #[cfg_attr(not(test_in_svsm), ignore = "Can only be run inside guest")]
317    fn spin_lock_irq_unsafe() {
318        let was_enabled = irqs_enabled();
319        raw_irqs_enable();
320
321        let spin_lock = SpinLock::new(0);
322        let guard = spin_lock.lock();
323        assert!(irqs_enabled());
324        drop(guard);
325        assert!(irqs_enabled());
326
327        if !was_enabled {
328            raw_irqs_disable();
329        }
330    }

351    fn spin_trylock_irq_safe() {
352        let was_enabled = irqs_enabled();
353        raw_irqs_enable();
354
355        let spin_lock = SpinLockIrqSafe::new(0);
356
357        // IRQs are enabled - taking the lock must succeed and disable IRQs
358        let g1 = spin_lock.try_lock();
359        assert!(g1.is_some());
360        assert!(irqs_disabled());
361
362        // Release lock and check if that enables IRQs
363        drop(g1);
364        assert!(irqs_enabled());
365
366        // Leave with IRQs configured as test was entered.
367        if !was_enabled {
368            raw_irqs_disable();
369        }
370    }

372    #[test]
373    #[cfg_attr(not(test_in_svsm), ignore = "Can only be run inside guest")]
374    fn spin_trylock_tpr() {
375        assert_eq!(raw_get_tpr(), 0);
376
377        let spin_lock = SpinLockTpr::new(0);

379        // TPR is zero - taking the lock must succeed and raise TPR.

383
384        // Release lock and check if that resets TPR.
