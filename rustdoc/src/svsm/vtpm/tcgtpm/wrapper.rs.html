32#[unsafe(no_mangle)]
33pub extern "C" fn malloc(size: c_ulong) -> *mut c_void {
34    if size == 0 {
35        return ptr::null_mut();
36    }
37
38    let Ok(layout) = layout_from_size(size as usize) else {
39        return ptr::null_mut();
40    };
41
42    // SAFETY: layout is guaranteed to be non-zero size. Memory may not be
43    // initiatlized, but that's what the caller expects.
44    unsafe { alloc(layout).cast() }
45}
46
47#[unsafe(no_mangle)]
48pub extern "C" fn calloc(items: c_ulong, size: c_ulong) -> *mut c_void {
49    let Some(new_size) = items.checked_mul(size) else {
50        return ptr::null_mut();
51    };
52
53    if new_size == 0 {
54        return ptr::null_mut();
55    }
56
57    let Ok(layout) = layout_from_size(new_size as usize) else {
58        return ptr::null_mut();
59    };
60
61    // SAFETY: layout is guaranteed to be non-zero size.
62    unsafe { alloc_zeroed(layout).cast() }
63}

65#[unsafe(no_mangle)]
66pub unsafe extern "C" fn realloc(p: *mut c_void, size: c_ulong) -> *mut c_void {
67    let ptr = p as *mut u8;
68    let new_size = size as usize;
69
70    if p.is_null() {
71        return malloc(size);
72    }
73
74    let Some(layout) = layout_from_ptr(ptr) else {
75        return ptr::null_mut();
76    };
77
78    if new_size == 0 {
79        // SAFETY: layout_from_ptr() call ensures that `ptr` was allocated
80        // with this allocator and we are using the same `layout` used to
81        // allocate `ptr`.
82        unsafe { dealloc(ptr, layout) };
83        return ptr::null_mut();
84    }
85
86    // This will fail if `new_size` rounded value exceeds `isize::MAX`
87    if Layout::from_size_align(new_size, layout.align()).is_err() {
88        return ptr::null_mut();
89    }
90
91    // SAFETY: layout_from_ptr() call ensures that `ptr` was allocated with
92    // this allocator and we are using the same `layout` used to allocate
93    // `ptr`. We also checked that `new_size` aligned does not overflow and
94    // it is not 0.
95    unsafe { _realloc(ptr, layout, new_size).cast() }
96}

98#[unsafe(no_mangle)]
99pub unsafe extern "C" fn free(p: *mut c_void) {
100    if p.is_null() {
101        return;
102    }
103    let ptr = p as *mut u8;
104    let Some(layout) = layout_from_ptr(ptr.cast()) else {
105        return;
106    };
107    // SAFETY: layout_from_ptr() call ensures that `ptr` was allocated
108    // with this allocator and we are using the same `layout` used to
109    // allocate `ptr`.
110    unsafe { dealloc(ptr, layout) }
111}

113#[unsafe(no_mangle)]
114pub unsafe extern "C" fn serial_out(s: *const c_char, size: c_int) {
115    // SAFETY: caller must provide safety requirements for
116    // [`core::slice::from_raw_parts`]
117    let str_slice: &[u8] = unsafe { from_raw_parts(s as *const u8, size as usize) };
118    if let Ok(rust_str) = from_utf8(str_slice) {
119        _print(format_args!("[SVSM] {rust_str}"));
120    } else {
121        log::error!("ERR: BUG: serial_out arg1 is not a valid utf8 string");
122    }
123}

130#[unsafe(no_mangle)]
131pub unsafe extern "C" fn fopen(path: *const c_char, mode: *const c_char) -> *mut c_void {
132    // SAFETY: caller must provide valid null-terminated C strings
133    let (path, mode) = unsafe { (CStr::from_ptr(path), CStr::from_ptr(mode)) };
134    let (Ok(path), Ok(mode)) = (path.to_str(), mode.to_str()) else {
135        return ptr::null_mut();
136    };
137
138    match CFile::fopen(path, mode) {
139        Some(nv) => Box::into_raw(Box::new(nv)).cast(),
140        None => ptr::null_mut(),
141    }
142}

144#[unsafe(no_mangle)]
145pub unsafe extern "C" fn fclose(file: *mut c_void) -> c_int {
146    if file.is_null() {
147        return -1;
148    }
149    // SAFETY: caller must pass a pointer previously returned by fopen
150    let mut nv = unsafe {
151        // Reclaim ownership so the CFile is dropped (and its buffer
152        // zeroized) when `nv` goes out of scope.
153        Box::from_raw(file.cast::<CFile>())
154    };
155    if nv.save().is_err() {
156        return -1;
157    }
158    0
159}

161#[unsafe(no_mangle)]
162pub unsafe extern "C" fn fread(
163    buf: *mut c_void,
164    size: usize,
165    count: usize,
166    file: *mut c_void,
167) -> usize {
168    // SAFETY: caller must pass a pointer previously returned by fopen
169    let Some(nv) = (unsafe { file.cast::<CFile>().as_mut() }) else {
170        return 0;
171    };
172    if buf.is_null() || size == 0 {
173        return 0;
174    }
175    let Some(total) = size.checked_mul(count) else {
176        return 0;
177    };
178    // SAFETY: buf is non-null (checked above); caller must guarantee
179    // it points to at least `size * count` valid bytes
180    let dst = unsafe { core::slice::from_raw_parts_mut(buf.cast::<u8>(), total) };
181    nv.fread(dst, size).unwrap_or(0)
182}

204    nv.fwrite(src, size).unwrap_or(0)

207#[unsafe(no_mangle)]
208pub unsafe extern "C" fn fseek(file: *mut c_void, offset: c_long, whence: c_int) -> c_int {
209    // SAFETY: caller must pass a pointer previously returned by fopen
210    let Some(nv) = (unsafe { file.cast::<CFile>().as_mut() }) else {
211        return -1;
212    };
213    if nv.fseek(offset as isize, whence).is_err() {
214        return -1;
215    }
216    0
217}

219#[unsafe(no_mangle)]
220pub unsafe extern "C" fn ftell(file: *mut c_void) -> c_long {
221    // SAFETY: caller must pass a pointer previously returned by fopen
222    let Some(nv) = (unsafe { file.cast::<CFile>().as_ref() }) else {
223        return -1;
224    };
225    match nv.ftell() {
226        Ok(pos) => pos.try_into().unwrap_or(-1),
227        Err(_) => -1,
228    }
229}

236    };
237    if nv.save().is_err() {
238        return -1;
